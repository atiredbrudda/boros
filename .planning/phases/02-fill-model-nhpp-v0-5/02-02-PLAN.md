```python
"""FILL-04: AMM consumption model (runtime analogue of DATA-05 ingest assertion).

Computes per-tick `amm_share(δ, t) = max(combined - raw, 0.0)` and asserts the
per-epoch runtime conservation identity.

Contract:
    book_ahead_end = book_ahead_start
                     - taker_consumption_ahead
                     - amm_consumption
                     + net_placements_ahead

Any violation raises AMMConsumptionIdentityError — must NOT be caught and
skipped. Defends PITFALLS §B3 at decision-engine call time (Phase 1 DATA-05 at
ingest time).
"""
from __future__ import annotations

import polars as pl
import structlog

from .schemas import AMMConsumptionIdentityError

log = structlog.get_logger(__name__)

# Re-export for convenience (Phase 4 DecisionEngine imports from here)
__all__ = [
    "AMMConsumptionIdentityError",
    "compute_amm_share_timeseries",
    "assert_epoch_conservation",
]


def compute_amm_share_timeseries(
    raw: pl.DataFrame,
    combined: pl.DataFrame,
    *,
    epoch_start_ts: int,
    epoch_end_ts: int,
) -> pl.DataFrame:
    """Compute amm_share(δ, t) = max(combined - raw, 0.0) per (snapshot_ts, tick).

    Filters both inputs to [epoch_start_ts, epoch_end_ts). Raises
    AMMConsumptionIdentityError if any (combined - raw) < -tolerance, mirroring
    the ingest-time assert_amm_conservation shape.

    Input schema (both):
        raw:      snapshot_ts | tick | size          (DATA-02 order_book_raw)
        combined: snapshot_ts | tick | size_combined (DATA-02 order_book_combined)

    Output schema:
        snapshot_ts | tick | amm_share
    """
    raw_sel = raw.filter(
        (pl.col("snapshot_ts") >= epoch_start_ts) & (pl.col("snapshot_ts") < epoch_end_ts)
    ).select(["snapshot_ts", "tick", "size"])
    combined_sel = combined.filter(
        (pl.col("snapshot_ts") >= epoch_start_ts) & (pl.col("snapshot_ts") < epoch_end_ts)
    ).select(["snapshot_ts", "tick", pl.col("size_combined")])

    joined = raw_sel.join(combined_sel, on=["snapshot_ts", "tick"], how="full", coalesce=True)
    joined = joined.with_columns(
        pl.col("size_combined").fill_null(0.0),
        pl.col("size").fill_null(0.0),
    )

    # Hard-fail on any negative combined-raw (below tolerance). Tolerance kept at 0
    # because Phase 1 DATA-05 ingest already rounded; any negative here is a real bug.
    neg = joined.filter(pl.col("size_combined") - pl.col("size") < -1e-9)
    if not neg.is_empty():
        log.error(
            "fill_model.amm_conservation_violation",
            epoch_start=epoch_start_ts,
            epoch_end=epoch_end_ts,
            offending_rows=neg.head(5).to_dicts(),
            n_offending=neg.height,
        )
        raise AMMConsumptionIdentityError(
            f"combined < raw on {neg.height} (snapshot_ts, tick) pairs during "
            f"[{epoch_start_ts}, {epoch_end_ts}); first 5: {neg.head(5).to_dicts()}"
        )

    return joined.with_columns(
        pl.max_horizontal(pl.col("size_combined") - pl.col("size"), pl.lit(0.0)).alias("amm_share")
    ).select(["snapshot_ts", "tick", "amm_share"])


def assert_epoch_conservation(
    book_ahead_start: float,
    book_ahead_end: float,
    taker_consumption_ahead: float,
    amm_consumption: float,
    net_placements_ahead: float,
    *,
    epoch_start_ts: int,
    epoch_end_ts: int,
    tolerance: float = 1e-6,
) -> None:
    """Runtime per-epoch flow identity. Raises on violation.

    book_ahead_end ≈ book_ahead_start
                      - taker_consumption_ahead
                      - amm_consumption
                      + net_placements_ahead
    """
    expected = (
        book_ahead_start - taker_consumption_ahead - amm_consumption + net_placements_ahead
    )
    delta = book_ahead_end - expected
    if abs(delta) > tolerance:
        log.error(
            "fill_model.amm_conservation_violation",
            epoch_start=epoch_start_ts,
            epoch_end=epoch_end_ts,
            identity_delta=float(delta),
            book_ahead_start=float(book_ahead_start),
            book_ahead_end=float(book_ahead_end),
            taker_consumption_ahead=float(taker_consumption_ahead),
            amm_consumption=float(amm_consumption),
            net_placements_ahead=float(net_placements_ahead),
        )
        raise AMMConsumptionIdentityError(
            f"per-epoch conservation identity violated: delta={delta:.6g} > tolerance={tolerance}, "
            f"epoch=[{epoch_start_ts}, {epoch_end_ts}]"
        )
```

```python
def _make_balanced_epoch(combined_at_tick: float = 100.0, raw_at_tick: float = 60.0) -> tuple[pl.DataFrame, pl.DataFrame]:

def test_amm_share_non_negative_basic() -> None:
```

```python
"""FILL-03: queue-position tracker with tri-track replay + drift diagnostic.

Replays the `market-trades` NDJSON tape (eventIndex-ordered) between hourly
`order_book_raw/` anchors, maintaining per-tick queue-ahead-of-us estimates in
three parallel tracks:
  - pessimistic: every invisible event treated as a new placement ahead of us
  - realized:    only visible market-trades consume the book
  - optimistic:  every invisible event treated as a cancel ahead of us

By construction (PITFALLS §B1): pessimistic >= realized >= optimistic at every
tick. The decision engine consumes the pessimistic bound via
`config.fill_model.queue_tracker_bias_mode = "pessimistic"` (default).

Drift is the diagnostic — widening bracket means more invisible-event activity.
NOT a failure — emit log events, let operator investigate.
"""
from __future__ import annotations

from typing import Literal

import numpy as np
import polars as pl
import structlog

from ..config.schema import BotConfig
from .amm_consumption import compute_amm_share_timeseries
from .schemas import ReplayResult

log = structlog.get_logger(__name__)

__all__ = ["QueueTracker", "compute_drift_diagnostic"]


def compute_drift_diagnostic(
    replay_queue: dict[int, float],
    snapshot_queue: dict[int, float],
) -> dict[str, float]:
    """Signed drift per tick: replay_predicted - snapshot_observed.

    Positive mean => replay over-estimates queue (conservative fill prediction).
    Negative mean => replay under-estimates (liberal). Both are information;
    neither is failure.

    Returns dict with keys: drift_signed_mean, drift_magnitude_mean,
    drift_std, n_ticks_compared.
    """
    ticks = set(replay_queue) | set(snapshot_queue)
    if not ticks:
        return {
            "drift_signed_mean": 0.0,
            "drift_magnitude_mean": 0.0,
            "drift_std": 0.0,
            "n_ticks_compared": 0.0,
        }
    drifts = np.array(
        [replay_queue.get(t, 0.0) - snapshot_queue.get(t, 0.0) for t in ticks],
        dtype=float,
    )
    return {
        "drift_signed_mean": float(drifts.mean()),
        "drift_magnitude_mean": float(np.abs(drifts).mean()),
        "drift_std": float(drifts.std(ddof=1)) if len(drifts) > 1 else 0.0,
        "n_ticks_compared": float(len(ticks)),
    }


class QueueTracker:
    """Tri-track replay per (market_id, side)."""

    def __init__(self, config: BotConfig, market_id: str, side: Literal["long", "short"]) -> None:
        self.config = config
        self.market_id = market_id
        self.side = side

    # ------------------------------------------------------------------
    # Internal replay helpers
    # ------------------------------------------------------------------
    def _load_trades(self, window_start_ts: int, window_end_ts: int) -> tuple[pl.DataFrame, str]:
        """Load + sort tape. Returns (sorted_df, event_index_coverage)."""
        trades = pl.scan_parquet("data/parquet/market_trades/**/*.parquet").filter(
            (pl.col("blockTimestamp") >= window_start_ts)
            & (pl.col("blockTimestamp") < window_end_ts)
            & (pl.col("market_id") == self.market_id)
            & (pl.col("side") == self.side)
        ).collect()

        coverage: str = "full"
        if trades.is_empty():
            return trades, coverage

        null_count = int(trades["eventIndex"].null_count())
        if null_count > 0:
            log.warning(
                "fill_model.eventindex_coverage_degraded",
                market_id=self.market_id,
                side=self.side,
                null_rows=null_count,
                total_rows=trades.height,
            )
            coverage = "degraded"
            trades = trades.with_columns(pl.col("eventIndex").fill_null(0))
        trades = trades.sort(["blockTimestamp", "eventIndex"])
        return trades, coverage

    def _load_raw_snapshots(self, window_start_ts: int, window_end_ts: int) -> pl.DataFrame:
        return pl.scan_parquet("data/parquet/order_book_raw/**/*.parquet").filter(
            (pl.col("snapshot_ts") >= window_start_ts)
            & (pl.col("snapshot_ts") < window_end_ts)
            & (pl.col("market_id") == self.market_id)
            & (pl.col("side") == self.side)
        ).collect()

    def _snapshot_queue_at(self, raw: pl.DataFrame, anchor_ts: int) -> dict[int, float]:
        """Tick -> queue size at a given snapshot_ts."""
        rows = raw.filter(pl.col("snapshot_ts") == anchor_ts).select(["tick", "size"]).to_dicts()
        return {int(r["tick"]): float(r["size"]) for r in rows}

    def _replay_realized(
        self,
        start_queue: dict[int, float],
        trades_in_interval: pl.DataFrame,
    ) -> dict[int, float]:
        """Decrement queue by visible takers only."""
        queue = dict(start_queue)
        for row in trades_in_interval.iter_rows(named=True):
            tick = int(row["tick"]) if "tick" in row else 0
            consumed = float(row["size"]) if "size" in row else 0.0
            queue[tick] = max(queue.get(tick, 0.0) - consumed, 0.0)
        return queue

    def _replay_pessimistic(
        self,
        start_queue: dict[int, float],
        trades_in_interval: pl.DataFrame,
        snapshot_next_queue: dict[int, float],
    ) -> dict[int, float]:
        """Invisible events = placements ahead. Take realized replay's result and
        add back any positive gap between snapshot_next and realized."""
        realized = self._replay_realized(start_queue, trades_in_interval)
        pessimistic = dict(realized)
        for tick, snap_size in snapshot_next_queue.items():
            if snap_size > realized.get(tick, 0.0):
                pessimistic[tick] = snap_size
        return pessimistic

    def _replay_optimistic(
        self,
        start_queue: dict[int, float],
        trades_in_interval: pl.DataFrame,
        snapshot_next_queue: dict[int, float],
    ) -> dict[int, float]:
        """Invisible events = cancels ahead. Take realized and cap at snapshot_next."""
        realized = self._replay_realized(start_queue, trades_in_interval)
        optimistic = dict(realized)
        for tick in realized:
            if snapshot_next_queue.get(tick, 0.0) < optimistic[tick]:
                optimistic[tick] = snapshot_next_queue.get(tick, 0.0)
        return optimistic

    # ------------------------------------------------------------------
    # Public entry point
    # ------------------------------------------------------------------
    def replay_window(self, window_start_ts: int, window_end_ts: int) -> ReplayResult:
        trades, coverage = self._load_trades(window_start_ts, window_end_ts)
        raw = self._load_raw_snapshots(window_start_ts, window_end_ts)
        anchors = sorted({int(t) for t in raw["snapshot_ts"].unique().to_list()})

        pess: dict[int, dict[int, float]] = {}
        real: dict[int, dict[int, float]] = {}
        opt: dict[int, dict[int, float]] = {}
        diags: list[dict[str, float]] = []

        if not anchors:
            return ReplayResult(
                market_id=self.market_id,
                side=self.side,
                window_start_ts=window_start_ts,
                window_end_ts=window_end_ts,
                event_index_coverage=coverage if coverage in {"full", "degraded"} else "full",  # type: ignore
            )

        # For each adjacent pair (anchor_i, anchor_{i+1}), run three replays.
        for i, anchor_ts in enumerate(anchors[:-1]):
            next_ts = anchors[i + 1]
            snap_now = self._snapshot_queue_at(raw, anchor_ts)
            snap_next = self._snapshot_queue_at(raw, next_ts)
            trades_iv = trades.filter(
                (pl.col("blockTimestamp") >= anchor_ts) & (pl.col("blockTimestamp") < next_ts)
            )

            real[next_ts] = self._replay_realized(snap_now, trades_iv)
            pess[next_ts] = self._replay_pessimistic(snap_now, trades_iv, snap_next)
            opt[next_ts] = self._replay_optimistic(snap_now, trades_iv, snap_next)

            # Drift diagnostic: replay_predicted (realized) vs actual snapshot at next_ts
            diag = compute_drift_diagnostic(replay_queue=real[next_ts], snapshot_queue=snap_next)
            log.info(
                "fill_model.queue_drift_reconciliation",
                market_id=self.market_id,
                side=self.side,
                anchor_ts=anchor_ts,
                next_ts=next_ts,
                drift_signed_mean=diag["drift_signed_mean"],
                drift_magnitude_mean=diag["drift_magnitude_mean"],
                drift_std=diag["drift_std"],
                n_ticks_compared=int(diag["n_ticks_compared"]),
            )
            diags.append(diag)

            # Bracket-width warn: if (pess - opt) > multiplier * hourly_mean_queue
            ticks = set(pess[next_ts]) | set(opt[next_ts])
            if ticks:
                widths = [pess[next_ts].get(t, 0.0) - opt[next_ts].get(t, 0.0) for t in ticks]
                mean_queue = float(np.mean([real[next_ts].get(t, 0.0) for t in ticks]) or 1.0)
                max_width = float(max(widths))
                multiplier = self.config.fill_model.bracket_width_warn_multiplier
                if mean_queue > 0.0 and max_width > multiplier * mean_queue:
                    log.warning(
                        "fill_model.bracket_width_warn",
                        market_id=self.market_id,
                        side=self.side,
                        anchor_ts=anchor_ts,
                        next_ts=next_ts,
                        bracket_width=max_width,
                        hourly_mean_queue=mean_queue,
                        multiplier=multiplier,
                    )

        return ReplayResult(
            market_id=self.market_id,
            side=self.side,
            window_start_ts=window_start_ts,
            window_end_ts=window_end_ts,
            pessimistic_queue_at_anchor=pess,
            realized_queue_at_anchor=real,
            optimistic_queue_at_anchor=opt,
            drift_diagnostics_per_hour=diags,
            event_index_coverage=coverage,  # type: ignore[arg-type]
        )
```

```markdown
# Queue tracker algorithm — v0.5

**Status:** Current — v0.5 regime
**Filed:** 2026-04-24
**Covers:** FILL-03 + FILL-06 fit-health scaffolding

## Three parallel replay tracks

Between adjacent hourly `order_book_raw` snapshots, replay the
`market-trades` NDJSON tape in `(blockTimestamp, eventIndex)` order with three
accounting schemes:

| Track | Treatment of invisible inter-snapshot events | Bound |
|-------|---------------------------------------------|------|
| **Pessimistic** | All invisible events = new placements ahead of us | High queue, low P(fill) |
| **Realized** | Only visible market-trades consume the book | Standard |
| **Optimistic** | All invisible events = cancels ahead of us | Low queue, high P(fill) |

**Invariant:** `pessimistic[tick] >= realized[tick] >= optimistic[tick]` for every tick. Violation = replay bug.

## Drift diagnostic

At each reconciliation, compute `drift[tick] = replay_predicted - snapshot_observed`. Aggregate:
- `drift_signed_mean` — positive means replay over-predicts queue (conservative)
- `drift_magnitude_mean` — average absolute drift
- `drift_std` — spread

Log event: `fill_model.queue_drift_reconciliation` (info level, per-hour).

## Bracket width warning

When `max_tick(pessimistic - optimistic) > bracket_width_warn_multiplier × hourly_mean_queue`:
emit `fill_model.bracket_width_warn` (warning level). Widening bracket IS information,
not failure — operator inspects. The fit can continue; the hour might be sample-weighted
down in future versions.

## Defaults

- `queue_tracker_bias_mode = "pessimistic"` — decision engine wants conservative queue
- `bracket_width_warn_multiplier = 0.5`

## eventIndex degradation path

If market-trades rows contain `eventIndex IS NULL` (per RESEARCH Assumption A3):
- Log `fill_model.eventindex_coverage_degraded` (warning)
- Fill nulls with 0 for sort determinism
- Return `ReplayResult.event_index_coverage = "degraded"`
- Downstream fit artifact marks the fit with same flag
- Do NOT crash — v0.5 remains shippable if a later re-screened market lacks eventIndex
```

```markdown
**Status:** Current — v0.5 regime
**Filed:** 2026-04-24
**Covers:** FILL-04

## Runtime analogue of DATA-05

Phase 1 `DATA-05` asserts at INGEST time: `combined_size(δ) >= raw_book_size(δ)`
for every tick in every snapshot (via `brain.data.ingester.order_book.assert_amm_conservation`).

Phase 2 adds a RUNTIME analogue: per-epoch flow identity.

### Per-tick amm_share

```
amm_share(δ, t) = max(combined(δ, t) - raw(δ, t), 0.0)
```

By construction `amm_share >= 0`. Any `combined < raw` beyond tolerance raises
`AMMConsumptionIdentityError` with the offending (snapshot_ts, tick) rows.

### Per-epoch flow identity

```
book_ahead_end = book_ahead_start
                  - taker_consumption_ahead
                  - amm_consumption
                  + net_placements_ahead
```

If `|delta| > 1e-6`, `assert_epoch_conservation(...)` logs
`fill_model.amm_conservation_violation` (error level) and raises.
```
