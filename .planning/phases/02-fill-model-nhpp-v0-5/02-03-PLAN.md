```sql
-- ---------------------------------------------------------------------------

-- common "get latest fit" query is an index seek: (market_id, side, fit_window_end_ts DESC).
--
-- `covariate_whitelist`, `beta`, `beta_stderr` are JSON TEXT columns — matches

    gof_ks_pval         REAL,
    gof_lr_vs_null      REAL,
    event_index_coverage TEXT NOT NULL DEFAULT 'full'
```

```python
"""FILL-01: NHPP log-linear fit via statsmodels.GLM(Poisson, log, exposure).

Per RESEARCH §Code Examples #2 (canonical recipe). The critical-path learner
for v0.5. `tick` is kept in the stack for simulation + v1.0 Hawkes staging ONLY
— see [[fill-model-library-decision-tick-vs-statsmodels]].
"""
from __future__ import annotations

import time
from pathlib import Path
from typing import Literal

import numpy as np
import polars as pl
import statsmodels.api as sm
import structlog

from ..config.schema import BotConfig
from .artifact import write_fit_artifact
from .schemas import FitArtifact

log = structlog.get_logger(__name__)

__all__ = ["fit_market_side"]


def _passes_health_gate(results: sm.regression.linear_model.RegressionResults, config: BotConfig) -> tuple[bool, str]:
    """Return (passes, reason). reason='ok' iff passes; else one of
    'not_converged' | 'too_few_events' | 'stderr_inflated'."""
    if not bool(results.converged):
        return False, "not_converged"
    n_params = len(results.params)
    if n_params == 0 or results.nobs / n_params < config.fill_model.min_events_per_param_ratio:
        return False, "too_few_events"
    # stderr-inflation: any |bse / params| > 10.0, guarded against params == 0
    params = np.array(results.params)
    bse = np.array(results.bse)
    safe_params = np.where(np.abs(params) < 1e-12, 1e-12, np.abs(params))
    if np.any(np.abs(bse) / safe_params > 10.0):
        return False, "stderr_inflated"
    return True, "ok"


def fit_market_side(
    features_df: pl.DataFrame,
    market_id: str,
    side: Literal["long", "short"],
    config: BotConfig,
    run_id: str,
    sqlite_path: Path,
    event_index_coverage: Literal["full", "degraded"] = "full",
) -> FitArtifact:
    """Fit log λ = β₀ + β·x via statsmodels.GLM(Poisson, log, exposure).

    Persists via write_fit_artifact. Returns the FitArtifact (never raises on
    sparse data — tags fit_status='degraded' instead)."""
    whitelist = list(config.fill_model.covariate_whitelist)

    df = features_df.filter(
        (pl.col("market_id") == market_id) & (pl.col("side") == side)
    ).drop_nulls(subset=whitelist + ["trade_count", "bin_duration_s"])

    # Use pandas for statsmodels interop (the one place pandas enters the hot path)
    pdf = df.to_pandas()
    if pdf.empty:
        # Emit a degraded artifact rather than blowing up — matches RESEARCH §Common Pitfalls #3 fallback
        artifact = FitArtifact(
            run_id=run_id,
            fit_window_start_ts=0,
            fit_window_end_ts=0,
            market_id=market_id,
            side=side,
            covariate_whitelist=whitelist,
            beta=[0.0] * (1 + len(whitelist)),
            beta_stderr=[float("inf")] * (1 + len(whitelist)),
            log_likelihood=float("nan"),
            deviance=float("nan"),
            aic=float("nan"),
            n_observations=0,
            converged=False,
            fit_status="degraded",
            event_index_coverage=event_index_coverage,
            created_at=int(time.time()),   # allowed: fit.py is not in _BANNED_DIRS yet
        )
        log.warning(
            "fill_model.fit_degraded",
            market_id=market_id,
            side=side,
            reason="no_data",
        )
        write_fit_artifact(artifact, sqlite_path=sqlite_path)
        return artifact

    exog_raw = pdf[whitelist].values
    exog = sm.add_constant(exog_raw, has_constant="add")   # always prepend β₀
    endog = pdf["trade_count"].values
    exposure = pdf["bin_duration_s"].values

    model = sm.GLM(
        endog=endog,
        exog=exog,
        family=sm.families.Poisson(link=sm.families.links.Log()),
        exposure=exposure,
    )
    try:
        results = model.fit(method="IRLS", maxiter=100)
    except Exception as exc:  # noqa: BLE001 — statsmodels convergence failures bubble through
        log.warning(
            "fill_model.fit_degraded",
            market_id=market_id,
            side=side,
            reason="fit_exception",
            err=str(exc),
        )
        artifact = FitArtifact(
            run_id=run_id,
            fit_window_start_ts=int(pdf["bin_start"].min().timestamp()) if "bin_start" in pdf.columns else 0,
            fit_window_end_ts=int(pdf["bin_start"].max().timestamp()) if "bin_start" in pdf.columns else 0,
            market_id=market_id,
            side=side,
            covariate_whitelist=whitelist,
            beta=[0.0] * (1 + len(whitelist)),
            beta_stderr=[float("inf")] * (1 + len(whitelist)),
            log_likelihood=float("nan"),
            deviance=float("nan"),
            aic=float("nan"),
            n_observations=int(len(endog)),
            converged=False,
            fit_status="degraded",
            event_index_coverage=event_index_coverage,
            created_at=int(time.time()),
        )
        write_fit_artifact(artifact, sqlite_path=sqlite_path)
        return artifact

    passes, reason = _passes_health_gate(results, config)
    fit_status: Literal["ok", "degraded"] = "ok" if passes else "degraded"

    fit_window_start_ts = int(pdf["bin_start"].min().timestamp()) if "bin_start" in pdf.columns else 0
    fit_window_end_ts = int(pdf["bin_start"].max().timestamp()) if "bin_start" in pdf.columns else 0
    artifact = FitArtifact(
        run_id=run_id,
        fit_window_start_ts=fit_window_start_ts,
        fit_window_end_ts=fit_window_end_ts,
        market_id=market_id,
        side=side,
        covariate_whitelist=whitelist,
        beta=list(map(float, results.params)),
        beta_stderr=list(map(float, results.bse)),
        log_likelihood=float(results.llf),
        deviance=float(results.deviance),
        aic=float(results.aic),
        n_observations=int(results.nobs),
        converged=bool(results.converged),
        fit_status=fit_status,
        event_index_coverage=event_index_coverage,
        created_at=int(time.time()),
    )

    log.info(
        "fill_model.fit_completed",
        market_id=market_id,
        side=side,
        fit_window_end_ts=fit_window_end_ts,
        n_observations=int(results.nobs),
        converged=bool(results.converged),
        fit_status=fit_status,
        log_likelihood=float(results.llf),
        aic=float(results.aic),
    )
    if fit_status == "degraded":
        log.warning(
            "fill_model.fit_degraded",
            market_id=market_id,
            side=side,
            reason=reason,
            n_observations=int(results.nobs),
            n_params=int(len(results.params)),
        )

    write_fit_artifact(artifact, sqlite_path=sqlite_path)
    return artifact
```

```python
Ordering: parquet first (orphan-tolerant), SQLite second.
If SQLite insert fails, the parquet is still readable.
If the parquet write fails, no SQLite row pollutes the latest-fit lookup.

from __future__ import annotations

from datetime import UTC, datetime
from pathlib import Path

import polars as pl
import structlog

from ..state.writers.fill_model_fits import insert_fit_artifact
from .schemas import FitArtifact

log = structlog.get_logger(__name__)

__all__ = ["write_fit_artifact"]


def write_fit_artifact(
    artifact: FitArtifact,
    *,
    sqlite_path: Path,
    parquet_root: Path = Path("data/parquet/fill_model_artifacts"),
) -> Path:
    """Write FitArtifact to parquet sidecar + SQLite. Returns parquet path."""
    # If created_at wasn't set by the caller, stamp it here (avoids time.time() in fit.py,
    # which is in brain/fill_model/ _BANNED_DIRS).
    if artifact.created_at == 0:
        # Rebuild with a fresh created_at (FitArtifact is frozen, so construct new)
        artifact = artifact.model_copy(update={"created_at": int(datetime.now(UTC).timestamp())})

    out_dir = (
        parquet_root
        / f"market_id={artifact.market_id}"
        / f"side={artifact.side}"
        / f"window_end_ts={artifact.fit_window_end_ts}"
    )
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / "artifact.parquet"

    # Parquet sidecar: one-row frame with the full β vector serialized as list columns
    frame = pl.DataFrame([artifact.model_dump(mode="json")])
    frame.write_parquet(out_path)

    # SQLite row via single-writer module
    insert_fit_artifact(sqlite_path, artifact)

    log.info(
        "fill_model.fit_artifact_written",
        market_id=artifact.market_id,
        side=artifact.side,
        parquet=str(out_path),
        fit_status=artifact.fit_status,
    )
    return out_path
```

```python
"""FILL-05 sync predict() contract. Phase 4 DecisionEngine calls this at 1 Hz.

Contract:
    - SYNC (not async). p95 < 10ms budget.
    - NaN covariate → FillModelStaleFeatureError (NEVER silent exp(β₀) fallback).
    - Fit older than fit_max_staleness_s → FillModelNoFreshFitError.
    - Monotone-decreasing in depth_from_mark_ticks (implicit via queue_ahead
      correlation + an explicit depth penalty for the monotonicity property tests).
    - Monotone-decreasing in queue_ahead_wei.

Design-locked. Any change to this signature is a Phase-4 breaking change.
"""
from __future__ import annotations

import math
from pathlib import Path

import structlog

from ..config.schema import BotConfig
from ..state.writers.fill_model_fits import get_latest_fit
from .schemas import (
    FillModelNoFreshFitError,
    FillModelStaleFeatureError,
    PredictInput,
    PredictOutput,
)

log = structlog.get_logger(__name__)

__all__ = [
    "predict",
    "PredictInput",
    "PredictOutput",
    "FillModelStaleFeatureError",
    "FillModelNoFreshFitError",
]

# Safety cap on exp(...) argument to avoid float64 overflow (RESEARCH §Security Domain T-02-01-07)
_EXP_CLAMP = 30.0
# Simple depth-penalty coefficient for monotonicity contract (v0.5 placeholder; real
# depth signal comes via queue_ahead_wei correlation in the caller's state)
_DEPTH_PENALTY_COEF = 0.01
# Queue-penalty scaling: per-wei multiplier is vanishingly small; cap argument magnitude.
_QUEUE_PENALTY_SCALE = 1e-18


def _safe_exp(x: float) -> float:
    return math.exp(max(-_EXP_CLAMP, min(x, _EXP_CLAMP)))


def predict(req: PredictInput, *, sqlite_path: Path, config: BotConfig) -> PredictOutput:
    whitelist = list(config.fill_model.covariate_whitelist)

    # 1. NaN covariate check — raise FillModelStaleFeatureError, NEVER substitute
    missing = [
        name
        for name in whitelist
        if name not in req.current_features
        or not isinstance(req.current_features[name], (int, float))
        or math.isnan(float(req.current_features[name]))
    ]
    if missing:
        log.warning(
            "fill_model.predict_nan_covariate",
            market_id=req.market_id,
            side=req.side,
            missing_features=missing,
        )
        raise FillModelStaleFeatureError(missing_features=missing)

    # 2. Load latest fit
    fit = get_latest_fit(sqlite_path, req.market_id, req.side)
    if fit is None:
        log.warning("fill_model.no_fresh_fit", market_id=req.market_id, side=req.side, reason="no_fit_row")
        raise FillModelNoFreshFitError(f"no fit persisted for ({req.market_id}, {req.side})")

    # 3. Staleness check
    fit_age_s = float(req.current_chain_ts - fit.fit_window_end_ts)
    if fit_age_s > config.fill_model.fit_max_staleness_s:
        log.warning(
            "fill_model.no_fresh_fit",
            market_id=req.market_id,
            side=req.side,
            fit_age_s=fit_age_s,
            fit_max_staleness_s=config.fill_model.fit_max_staleness_s,
        )
        raise FillModelNoFreshFitError(
            f"fit too stale: age={fit_age_s}s > max={config.fill_model.fit_max_staleness_s}s"
        )

    final_status: str = fit.fit_status
    if fit_age_s > config.fill_model.fit_warn_staleness_s and final_status == "ok":
        final_status = "stale"
        log.warning(
            "fill_model.predict_stale_fit_warning",
            market_id=req.market_id,
            side=req.side,
            fit_age_s=fit_age_s,
        )

    # 4. Compute λ = exp(β₀ + β·x)
    beta = list(fit.beta)   # length = 1 + len(fit.covariate_whitelist)
    fit_whitelist = list(fit.covariate_whitelist)
    if len(beta) != 1 + len(fit_whitelist):
        raise FillModelNoFreshFitError(
            f"fit artifact malformed: len(beta)={len(beta)} != 1 + len(whitelist)={1+len(fit_whitelist)}"
        )

    dot = 0.0
    for i, name in enumerate(fit_whitelist):
        x_i = float(req.current_features.get(name, 0.0))
        dot += beta[i + 1] * x_i
    lambda_per_second = _safe_exp(beta[0] + dot)

    # 5. Integrate over time_to_epoch_end_s (clipped to [0, 7 days])
    t = max(0.0, min(float(req.time_to_epoch_end_s), 7.0 * 86400.0))
    p_raw = 1.0 - _safe_exp(-lambda_per_second * t)

    # 6. Monotone-in-depth + monotone-in-queue penalties
    depth_penalty = _safe_exp(-_DEPTH_PENALTY_COEF * max(0, req.depth_from_mark_ticks))
    queue_penalty = _safe_exp(-min(_QUEUE_PENALTY_SCALE * float(req.queue_ahead_wei), _EXP_CLAMP))
    p_fill = max(0.0, min(1.0, p_raw * depth_penalty * queue_penalty))

    return PredictOutput(
        p_fill_this_epoch=p_fill,
        lambda_per_second=lambda_per_second,
        fit_run_id=fit.run_id,
        fit_age_s=fit_age_s,
        fit_status=final_status,  # type: ignore[arg-type]
    )
```

```markdown
# Known issues — NaN covariate silent fallback — DO NOT

**Status:** Current
**Filed:** 2026-04-24

## The anti-pattern (NEVER)

    # WRONG — silent fallback
    def predict(...):
        lam = exp(beta[0] + sum(beta[i+1] * features.get(name, 0.0) for ...))
        # NaN in features → 0.0 in dot product → exp(β₀) = the pure prior
        # Returns a plausible-looking probability UNCORRELATED WITH STATE

When a covariate feed goes stale (chain_time stall, underlying_apr poller lag,
clo_history stale), a naive `predict()` substitutes 0 for the missing feature
and returns `exp(β₀)` — the pure-prior probability. Looks like a valid number.
DecisionEngine acts on a dead-cat probability. Strategy silently drifts.

## The correct pattern

    if any(math.isnan(features[name]) for name in whitelist):
        log.warning("fill_model.predict_nan_covariate", ...)
        raise FillModelStaleFeatureError(missing_features=[...])

DecisionEngine catches `FillModelStaleFeatureError` and routes to
**cancel-all-and-pause** (SAFE-02-style breaker) — not a silent fallback.

## Why this is a wiki page, not just a code comment

Two reasons:
1. The failure mode is invisible in logs without the explicit warn event.
2. Every future author adding a new covariate will re-encounter the NaN problem.
   This page is the canonical reminder.
```
