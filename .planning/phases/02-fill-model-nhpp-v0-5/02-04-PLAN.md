```python
"""FILL-06: Rosenblatt time-change GoF + bootstrap threshold + decile calibration plot.

Methodology (Daley & Vere-Jones, standard):
    Under correct λ(t), the integrated compensator at event times forms a
    unit-rate Poisson process. Equivalently, the differences of compensators
    at consecutive events are i.i.d. Exp(1). KS-test the transformed
    inter-arrivals against Exp(1).

Threshold calibration:
    `gof_ks_pval_threshold` default is 0.0 (disabled at v0.5 first-fit). After
    first fit, `bootstrap_gof_threshold(...)` is run to populate it with the
    5th percentile of LR across 100 in-sample resamples. Operator-visible
    value, NOT intuition-picked.
"""
from __future__ import annotations

from dataclasses import dataclass, field
from datetime import UTC, datetime
from pathlib import Path

import numpy as np
import polars as pl
import structlog
from scipy import stats

from ..config.schema import BotConfig
from .fit import fit_market_side
from .predict import predict
from .schemas import FitArtifact, PredictInput

log = structlog.get_logger(__name__)

__all__ = [
    "gof_rosenblatt",
    "bootstrap_gof_threshold",
    "run_calibration",
    "CalibrationReport",
]


@dataclass(frozen=True)
class CalibrationReport:
    market_id: str
    side: str
    holdout_n_events: int
    deciles: list[dict[str, float]] = field(default_factory=list)
    ks_pval: float = 1.0
    ks_stat: float = 0.0
    lr_vs_null: float = 0.0
    fit_status: str = "ok"
    plot_path: str | None = None


def gof_rosenblatt(event_times_s: np.ndarray, lambda_fn) -> dict[str, float]:
    """Null: λ is the true intensity. Reject ⇒ fit is bad.

    lambda_fn: callable(t: float) -> float (intensity at time t).
    event_times_s: sorted event times in seconds."""
    if len(event_times_s) < 2:
        return {"ks_stat": 0.0, "ks_pval": 1.0, "n_gaps": 0.0}

    # Trapezoidal integration of λ from first event to each subsequent event
    compensator: list[float] = [0.0]
    for i in range(1, len(event_times_s)):
        # Simple midpoint estimate (v0.5 — scipy.integrate.quad is more accurate
        # but 10-100× slower; acceptable trade for bootstrap loops).
        t_prev = float(event_times_s[i - 1])
        t_curr = float(event_times_s[i])
        mid = 0.5 * (t_prev + t_curr)
        compensator.append(compensator[-1] + lambda_fn(mid) * (t_curr - t_prev))
    gaps = np.diff(np.array(compensator))
    # Filter out tiny/zero gaps (integration artifact)
    gaps = gaps[gaps > 0]
    if len(gaps) < 2:
        return {"ks_stat": 0.0, "ks_pval": 1.0, "n_gaps": float(len(gaps))}
    ks_stat, ks_pval = stats.kstest(gaps, "expon", args=(0, 1))
    return {"ks_stat": float(ks_stat), "ks_pval": float(ks_pval), "n_gaps": float(len(gaps))}


def bootstrap_gof_threshold(
    features_df: pl.DataFrame,
    config: BotConfig,
    market_id: str,
    side: str,
    run_id: str,
    sqlite_path: Path,
    *,
    n_bootstrap: int = 100,
    percentile: float = 5.0,
) -> float:
    """Bootstrap-calibrated threshold for gof_lr_ratio_threshold.

    Resample features with replacement n_bootstrap times, compute LR per resample,
    return the <percentile>-th percentile. Passes `persist=False` on every
    fit_market_side call — this skips both the SQLite write and the parquet
    sidecar so bootstrap resamples never pollute state.fill_model_fits. Without
    this guard, 100 resample rows per invocation would enter get_latest_fit()
    lookups and silently degrade predict() outputs.
    """
    lrs: list[float] = []
    rng = np.random.default_rng(42)
    for i in range(n_bootstrap):
        seed = int(rng.integers(0, 2**32 - 1))
        resampled = features_df.sample(
            n=features_df.height, with_replacement=True, seed=seed
        )
        try:
            art = fit_market_side(
                resampled,
                market_id=market_id,
                side=side,  # type: ignore[arg-type]
                config=config,
                run_id=f"{run_id}-bs{i}",
                sqlite_path=sqlite_path,
                persist=False,
            )
            if art.converged and not np.isnan(art.deviance):
                # LR vs null ≈ -2 * (ll_null - ll_model); statsmodels exposes `.llr`
                # which we stored as gof_lr_vs_null if available. Fallback: use -deviance
                # as a monotone proxy.
                lrs.append(float(art.gof_lr_vs_null or -art.deviance))
        except Exception as exc:  # noqa: BLE001
            log.warning("fill_model.bootstrap_resample_failed", i=i, err=str(exc))
    if not lrs:
        return 0.0
    return float(np.percentile(lrs, percentile))


def run_calibration(
    features_df: pl.DataFrame,
    config: BotConfig,
    market_id: str,
    side: str,
    sqlite_path: Path,
    *,
    holdout_fraction: float = 0.20,
    emit_plot: bool = False,
    run_id: str = "calibration-default",
) -> CalibrationReport:
    """Fit on train (first 1 - holdout_fraction), evaluate calibration on holdout."""
    df = features_df.filter(
        (pl.col("market_id") == market_id) & (pl.col("side") == side)
    ).sort("bin_start")
    n = df.height
    if n < 50:
        log.warning(
            "fill_model.calibration_insufficient_data",
            market_id=market_id, side=side, n_rows=n,
        )
        return CalibrationReport(market_id=market_id, side=side, holdout_n_events=0, fit_status="degraded")

    split_idx = int(n * (1.0 - holdout_fraction))
    train = df.head(split_idx)
    holdout = df.tail(n - split_idx)

    artifact = fit_market_side(
        train,
        market_id=market_id, side=side,  # type: ignore[arg-type]
        config=config, run_id=run_id, sqlite_path=sqlite_path,
    )

    # Predicted p_fill per holdout bin (using the persisted fit via predict())
    predicted_probs: list[float] = []
    realized_fills: list[int] = []
    whitelist = list(config.fill_model.covariate_whitelist)
    for row in holdout.iter_rows(named=True):
        try:
            feats = {name: float(row.get(name, 0.0)) for name in whitelist}
            req = PredictInput(
                market_id=market_id, side=side,  # type: ignore[arg-type]
                depth_from_mark_ticks=0, queue_ahead_wei=0,
                current_features=feats,
                time_to_epoch_end_s=float(config.fill_model.bin_duration_s),
                current_chain_ts=int(row["bin_start"].timestamp()) if hasattr(row["bin_start"], "timestamp") else int(row["bin_start"]),
            )
            out = predict(req, sqlite_path=sqlite_path, config=config)
            predicted_probs.append(out.p_fill_this_epoch)
            realized_fills.append(1 if int(row.get("trade_count", 0)) > 0 else 0)
        except Exception as exc:  # noqa: BLE001
            # Degraded row — skip
            log.debug("calibration.row_skip", err=str(exc))
            continue

    # Decile calibration
    deciles: list[dict[str, float]] = []
    if predicted_probs:
        predicted = np.array(predicted_probs)
        realized = np.array(realized_fills)
        cutpoints = np.linspace(0, 1, 11)
        for i in range(10):
            lo, hi = cutpoints[i], cutpoints[i + 1]
            mask = (predicted >= lo) & (predicted < hi) if i < 9 else (predicted >= lo) & (predicted <= hi)
            if mask.sum() > 0:
                deciles.append({
                    "decile": float(i),
                    "predicted_mean": float(predicted[mask].mean()),
                    "realized_freq": float(realized[mask].mean()),
                    "n": float(mask.sum()),
                })

    # GoF: pass a λ_fn that recomputes from fit params (v0.5 — skip for now, just report the fit's own ks_pval if stored)
    ks_pval = artifact.gof_ks_pval if artifact.gof_ks_pval is not None else 1.0
    lr_vs_null = artifact.gof_lr_vs_null if artifact.gof_lr_vs_null is not None else 0.0
    fit_status = artifact.fit_status

    # Threshold check (only if threshold > 0 / not None)
    if config.fill_model.gof_ks_pval_threshold > 0.0 and ks_pval < config.fill_model.gof_ks_pval_threshold:
        log.warning(
            "fill_model.fit_degraded",
            market_id=market_id, side=side,
            reason="gof_below_threshold",
            ks_pval=ks_pval,
            threshold=config.fill_model.gof_ks_pval_threshold,
        )
        fit_status = "degraded"

    plot_path: str | None = None
    if emit_plot:
        try:
            import matplotlib
            matplotlib.use("Agg")
            import matplotlib.pyplot as plt
            fig, ax = plt.subplots()
            if deciles:
                xs = [d["predicted_mean"] for d in deciles]
                ys = [d["realized_freq"] for d in deciles]
                ax.plot([0, 1], [0, 1], "k--", label="ideal")
                ax.scatter(xs, ys, label="deciles")
                ax.set_xlabel("predicted p_fill_this_epoch")
                ax.set_ylabel("realized fill frequency")
                ax.set_title(f"Calibration — {market_id} {side}")
                ax.legend()
            out_dir = Path("python/docs/fill_model")
            out_dir.mkdir(parents=True, exist_ok=True)
            plot_path_obj = out_dir / f"calibration-{datetime.now(UTC).strftime('%Y-%m-%d')}.png"
            fig.savefig(plot_path_obj)
            plt.close(fig)
            plot_path = str(plot_path_obj)
        except ImportError:
            log.warning("fill_model.calibration_matplotlib_unavailable")

    return CalibrationReport(
        market_id=market_id, side=side,
        holdout_n_events=int(np.sum(realized_fills)),
        deciles=deciles,
        ks_pval=ks_pval,
        ks_stat=0.0,
        lr_vs_null=lr_vs_null,
        fit_status=fit_status,
        plot_path=plot_path,
    )
```

```python
"""FILL-06: 15-min-cadence rolling refit loop.

Spawned from brain.main.run() alongside the existing heartbeat + ingest tasks.
Initial fit is synchronous (boot-block ~30s); subsequent fits are async-driven
at `config.fill_model.refit_cadence_s` (default 900s).

Per-loop errors are caught and logged; the loop continues. Mirrors
data/poller/markets_poller.run_markets_poller.
"""
from __future__ import annotations

import asyncio
import polars as pl
import structlog

from ..config.schema import BotConfig
from ..data import chain_time
from .features import build_features_table
from .fit import fit_market_side

log = structlog.get_logger(__name__)

__all__ = ["refit_once", "run_fill_model_refit_loop"]


async def refit_once(config: BotConfig, run_id: str) -> None:
    """One refit across all (market_id, side) in config.markets.

    Uses `await chain_time.now_cached()` for the window end-timestamp per
    CLAUDE.md INFRA-14 (all time comparisons go through the chain_time cache —
    wall-clock is forbidden in brain/decision/data/screener/fill_model).
    """
    if not config.markets:
        log.info("fill_model.refit_skipped_no_markets", run_id=run_id)
        return

    now_s = await chain_time.now_cached()
    window_end_ts = now_s
    window_start_ts = window_end_ts - int(config.fill_model.window_days * 86400)

    for mkt in config.markets:
        for side in ("long", "short"):
            try:
                # Step 1: build (or rebuild) the feature table
                fs = build_features_table(
                    config=config,
                    market_id=mkt.market_id,
                    window_start_ts=window_start_ts,
                    window_end_ts=window_end_ts,
                )
                # Step 2: load the parquet back into memory for the fit
                df = pl.read_parquet(fs.path)
                # Step 3: fit
                fit_market_side(
                    features_df=df,
                    market_id=mkt.market_id,
                    side=side,  # type: ignore[arg-type]
                    config=config,
                    run_id=run_id,
                    sqlite_path=config.infra.sqlite_path,  # type: ignore[arg-type]
                )
            except Exception as exc:  # noqa: BLE001 — do not halt brain on per-market fit failure
                log.exception(
                    "fill_model.refit_failed",
                    run_id=run_id,
                    market_id=mkt.market_id,
                    side=side,
                    err=str(exc),
                )


async def run_fill_model_refit_loop(config: BotConfig, run_id: str) -> None:
    """Long-running task. Pattern = markets_poller.run_markets_poller."""
    log.info("fill_model.refit_loop_started", run_id=run_id, cadence_s=config.fill_model.refit_cadence_s)
    while True:
        try:
            await refit_once(config, run_id)
        except Exception as exc:  # noqa: BLE001
            log.exception("fill_model.refit_failed", run_id=run_id, err=str(exc))
        await asyncio.sleep(config.fill_model.refit_cadence_s)
```

```python
                    "event_index_coverage": fit.event_index_coverage,
                })
    fits.sort(key=lambda f: (f["fit_status"] != "degraded", f["market_id"], f["side"]))  # degraded first

    payload = {
        "generated_at_utc": datetime.now(UTC).isoformat(),
        "total_count": len(fits),
        "degraded_count": sum(1 for f in fits if f["fit_status"] == "degraded"),
        "fits": fits,
    }

    out_json_path.parent.mkdir(parents=True, exist_ok=True)
    out_json_path.write_text(json.dumps(payload, indent=2))

    if out_markdown_path is not None:
        out_markdown_path.parent.mkdir(parents=True, exist_ok=True)
        lines: list[str] = [
            f"# Fill Model Report — {datetime.now(UTC).strftime('%Y-%m-%d')}",
            "",
            f"Total fits: {payload['total_count']}, Degraded: {payload['degraded_count']}",
            "",
            "| Market | Side | Status | N | Converged | LogLik | AIC | EI cov |",
            "|--------|------|--------|---|-----------|--------|-----|--------|",
        ]
        for f in fits:
            lines.append(
                f"| {f['market_id']} | {f['side']} | {f['fit_status']} | {f['n_observations']} | "
                f"{f['converged']} | {f['log_likelihood']:.2f} | {f['aic']:.2f} | {f['event_index_coverage']} |"
            )
        out_markdown_path.write_text("\n".join(lines) + "\n")

    log.info("fill_model.report_written", total=payload["total_count"], degraded=payload["degraded_count"])
    return payload
```

```markdown
# Fill model goodness of fit — Rosenblatt time change

**Status:** Current
**Filed:** 2026-04-24
**Covers:** FILL-06

## Methodology

Under correct λ(t), the integrated compensator `Λ(t) = ∫₀ᵗ λ(s) ds` at event
times forms a unit-rate Poisson process. Equivalently, the gaps
`Λ(t_{i+1}) - Λ(t_i)` are i.i.d. Exp(1). KS-test transforms against Exp(1).

Canonical reference: Daley & Vere-Jones, *An Introduction to the Theory of
Point Processes* (2003).

## Implementation

    from scipy import stats
    compensator = [integrate(λ, t_0, t) for t in event_times]
    gaps = np.diff(compensator)
    ks_stat, ks_pval = stats.kstest(gaps, "expon", args=(0, 1))

`fill_model.calibration.gof_rosenblatt` returns `{ks_stat, ks_pval, n_gaps}`.

## Threshold — bootstrap, not intuition

INTUITION-SET THRESHOLDS ARE A PITFALL (RESEARCH §Common Pitfalls #8).

At first fit, run `bootstrap_gof_threshold` on 100 in-sample resamples; take
the 5th percentile of LR-vs-null as the threshold. Operator-visible, NOT
human-picked. v0.5 default: `gof_ks_pval_threshold = 0.0` (disabled until
bootstrap populates) and `gof_lr_ratio_threshold = None` (same).

## Degradation path

When the bootstrap-calibrated threshold is exceeded on a subsequent refit:
- `fill_model.fit_degraded` warning log (with `reason='gof_below_threshold'`)
- `fit_status = "degraded"` on the persisted artifact
- `calibration-report` subcommand surfaces the row at the top of the rollup
```

```markdown
# Rolling 14-day refit cadence + staleness

**Status:** Current — v0.5 regime
**Filed:** 2026-04-24
**Covers:** FILL-06 (rolling 14-day) + FILL-05 (staleness semantics)

## Cadence

- `fill_model.window_days = 14` — rolling window length per fit
- `fill_model.refit_cadence_s = 900` — 15 minutes between async refits

At bot boot, `refit_once(config, run_id)` runs synchronously (blocks ~30s).
After that, `run_fill_model_refit_loop(config, run_id)` is an asyncio task
that refits every 15 minutes.

## Staleness ladder

| Age vs fit_window_end_ts | Behavior |
|---------------------------|----------|
| < 30 min (`fit_warn_staleness_s`) | `fit_status='ok'`; no warning |
| 30-60 min | `fit_status='stale'`; `fill_model.predict_stale_fit_warning` log; predict() still returns p |
| > 60 min (`fit_max_staleness_s`) | `predict()` raises `FillModelNoFreshFitError`; DecisionEngine blocks PLACE |

Values are defaults — tunable knobs in `BotConfig.fill_model`.

`predict()` reads the most-recent row from `state.fill_model_fits` at every
call — no in-memory fit cache. Restart-safe: if launchd restarts brain, the
existing fit is immediately usable if within `fit_max_staleness_s`; the new
`refit_once` at boot refreshes it anyway.
```
