class MarketRewardCalibration(StrictBase):
    """LOCKED D-01 (2026-04-24): operator-manual per-market reward-budget calibration.

    Refreshed at daily re-screen. data_source='operator' lineage marker when rows
    are also mirrored into the new state.market_reward_calibration SQLite table
    (see python/src/boros_brain/state/writers/market_reward_calibration.py).

    Replaces the Phase 1 UAT ELiqPoller carried concern — liquidityRewardPerEpoch,
    longBudget, shortBudget are NOT in the live /open-api/v1/markets response;
    the operator pastes values from Pendle docs / dApp UI / team announcements.
    """
    e_liq_per_epoch_pendle: float = Field(..., ge=0.0)
    long_side_split: float = Field(..., ge=0.0, le=1.0)
    short_side_split: float = Field(..., ge=0.0, le=1.0)
    calibrated_at_utc: str   # ISO8601
    source: Literal["dapp_ui", "team_statement", "pendle_docs"] = "dapp_ui"

    @field_validator("short_side_split")
    @classmethod
    def _splits_sum_to_one(cls, v: float, info) -> float:
        long = info.data.get("long_side_split")
        if long is not None and abs(long + v - 1.0) > 1e-9:
            raise ValueError(
                f"long_side_split + short_side_split must == 1.0 (got {long} + {v} = {long + v}). "
                "P3-4 operator-error guard."
            )
        return v

"""Single writer for competition_history (brain-owned). PROTO-02 discipline.

data_source enum: 'live' (REST + parquet both healthy) or 'fallback_parquet_only'

from __future__ import annotations
import sqlite3
from pathlib import Path
from typing import Literal

DataSource = Literal["live", "fallback_parquet_only"]
_ALLOWED: frozenset[str] = frozenset({"live", "fallback_parquet_only"})

__all__ = ["insert_competition_row", "get_latest_competition"]


def insert_competition_row(
    conn: sqlite3.Connection,
    *,
    polled_at_utc: str,
    chain_ts: int | None,
    market_id: str,
    q_me_pendle: float,
    sigma_others_pendle: float,
    run_id: str,
    data_source: DataSource = "live",
) -> None:
    if data_source not in _ALLOWED:
        raise ValueError(f"data_source={data_source!r} not in {sorted(_ALLOWED)}")
    conn.execute(
        "INSERT INTO competition_history "
        "(polled_at_utc, chain_ts, market_id, q_me_pendle, sigma_others_pendle, data_source, run_id) "
        "VALUES (?, ?, ?, ?, ?, ?, ?)",
        (polled_at_utc, chain_ts, market_id, q_me_pendle, sigma_others_pendle, data_source, run_id),
    )
    # Caller owns the transaction — NO conn.commit() here (db.py contract).


def get_latest_competition(sqlite_path: Path, market_id: str):
    """Return most-recent competition_history row for market_id, or None."""
    from boros_brain.state.db import open_connection
    with open_connection(sqlite_path) as conn:
        row = conn.execute(
            "SELECT polled_at_utc, chain_ts, market_id, q_me_pendle, sigma_others_pendle, data_source, run_id "

"""Single writer for market_reward_calibration (brain-owned). LOCKED D-01 2026-04-24.

from __future__ import annotations
import sqlite3
from pathlib import Path
from typing import Literal

Source = Literal["dapp_ui", "team_statement", "pendle_docs"]
_ALLOWED_SOURCES: frozenset[str] = frozenset({"dapp_ui", "team_statement", "pendle_docs"})

__all__ = ["insert_market_reward_calibration", "get_latest_market_reward_calibration"]


def insert_market_reward_calibration(
    sqlite_path: Path,
    *,
    market_id: str,
    e_liq_per_epoch_pendle: float,
    long_side_split: float,
    short_side_split: float,
    calibrated_at_utc: str,
    source: Source = "dapp_ui",
    run_id: str,
) -> None:
    """Self-contained single-writer — opens its own connection.

    Raises ValueError BEFORE SQL round-trip if splits don't sum to 1.0 (P3-4 guard)
    or source is not in allowed enum. SQL CHECK constraints are belt-and-braces.
    """
    if source not in _ALLOWED_SOURCES:
        raise ValueError(f"source={source!r} not in {sorted(_ALLOWED_SOURCES)}")
    if abs(long_side_split + short_side_split - 1.0) > 1e-9:
        raise ValueError(
            f"long_side_split + short_side_split must == 1.0 "
            f"(got {long_side_split} + {short_side_split}). P3-4 guard."
        )
    if e_liq_per_epoch_pendle < 0.0:
        raise ValueError(f"e_liq_per_epoch_pendle must be >= 0.0, got {e_liq_per_epoch_pendle}")

    from boros_brain.state.db import open_connection
    import structlog
    log = structlog.get_logger(__name__)

    with open_connection(sqlite_path) as conn:
        conn.execute(
            "INSERT INTO market_reward_calibration "
            "(market_id, e_liq_per_epoch_pendle, long_side_split, short_side_split, "
            " calibrated_at_utc, data_source, source, run_id) "
            "VALUES (?, ?, ?, ?, ?, 'operator', ?, ?)",
            (market_id, e_liq_per_epoch_pendle, long_side_split, short_side_split,
             calibrated_at_utc, source, run_id),
        )
    log.info(
        "reward_model.calibration_persisted",
        market_id=market_id,
        e_liq_per_epoch_pendle=e_liq_per_epoch_pendle,
        long_side_split=long_side_split,
        short_side_split=short_side_split,
        source=source,
        run_id=run_id,
    )


def get_latest_market_reward_calibration(sqlite_path: Path, market_id: str):

```python
"""REWARD-01 closed-form capped pro-rata reward-share formula.

Verbatim from boros_consolidated_plan.md §4 step 4. PENDLE-unit discipline.
Fees are NOT subtracted here — they're absorbed inside LossModel.expected_loss
(see [[fee-cost-term-in-expected-loss]]). This matches the EV-rule derivation
where R_capped is the revenue side and E[loss|fill] × P(fill) is the cost side.

PITFALLS §A4 absorption: returns three-way `regime` classification ('sub_cap',
'cap_binding', 'market_saturated') alongside the capped share. Kinked-curve
geometry is thus observable to Phase 4 DecisionEngine.
"""
from __future__ import annotations

import structlog

from ..screener.economics_gate import EPOCHS_PER_YEAR  # noqa: F401 — reused pattern (may inform docs), but our formula uses epoch_days directly
from .schemas import RewardCurveInput, RewardCurveOutput

log = structlog.get_logger(__name__)

__all__ = ["r_capped_curve"]


def r_capped_curve(req: RewardCurveInput) -> RewardCurveOutput:
    """Compute R_uncapped, R_capped, cap_budget, and regime label.

    Formula:
        R_uncapped = E_liq × side_split × (Q_me × t_frac) / (Σ_others + Q_me × t_frac)
        cap_budget = 10 × collateral_usd × (epoch_days / 365) / pendle_price_usd
        R_capped   = min(R_uncapped, cap_budget)

    Regimes:
        - cap_binding: R_uncapped >= cap_budget (the cap clips the curve)
        - market_saturated: Σ_others == 0 AND Q_me > 0 (we have the market to ourselves)
        - sub_cap: otherwise (the healthy regime at $100 collateral)

    Units: all outputs in PENDLE.
    """
    # Input validation (fail closed on operator / caller errors)
    if req.pendle_price_usd <= 0.0:
        raise ValueError(f"pendle_price_usd must be > 0, got {req.pendle_price_usd}")
    if req.e_liq_per_epoch_pendle < 0.0:
        raise ValueError(f"e_liq_per_epoch_pendle must be >= 0.0, got {req.e_liq_per_epoch_pendle}")
    if req.epoch_days <= 0.0:
        raise ValueError(f"epoch_days must be > 0, got {req.epoch_days}")

    # r_uncapped with div-by-zero guard (T-03-03 mitigation)
    denom = req.sigma_others_pendle + req.q_me_pendle * req.t_frac
    if denom == 0.0 or req.q_me_pendle == 0.0:
        r_uncapped = 0.0
    else:
        r_uncapped = (
            req.e_liq_per_epoch_pendle
            * req.side_split
            * (req.q_me_pendle * req.t_frac)
            / denom
        )

    # cap_budget_per_epoch in PENDLE-units (A4 kink point)
    cap_budget = 10.0 * req.collateral_usd * (req.epoch_days / 365.0) / req.pendle_price_usd

    # regime classification
    if req.sigma_others_pendle == 0.0 and req.q_me_pendle > 0.0:
        regime = "market_saturated"
    elif r_uncapped >= cap_budget:
        regime = "cap_binding"
    else:
        regime = "sub_cap"

    r_capped = min(r_uncapped, cap_budget)

    return RewardCurveOutput(
        r_uncapped_pendle=r_uncapped,
        r_capped_pendle=r_capped,
        cap_budget_pendle=cap_budget,
        regime=regime,
    )
```

**Step 2 — Create `python/src/boros_brain/reward_model/q_cap.py`:**

```python
"""REWARD-04 closed-form Q_cap algebra.

Boundary where R_uncapped(Q_me) first hits cap_budget. From consolidated plan §4 step 4:

    Q_cap = Σ_others × cap_budget / (E_liq × side_split × t_frac - cap_budget × t_frac)

Returns math.inf when cap is algebraically unreachable at any finite Q_me (denominator <= 0).

At $100 collateral on ETHUSDT-29MAY2026 anchor (cap_budget ≈ 0.274 PENDLE, E_liq × side_split ≈ 234 PENDLE),
Q_cap is finite but LARGER than the collateral-constrained Q_max at $100, so it never practically binds.
At $5000 collateral (cap_budget ≈ 13.7 PENDLE), Q_cap is smaller AND less than Q_max — cap binds.
"""
from __future__ import annotations

import math

__all__ = ["compute_q_cap"]


def compute_q_cap(
    *,
    sigma_others_pendle: float,
    e_liq_per_epoch_pendle: float,
    side_split: float,
    t_frac: float,
    collateral_usd: float,
    pendle_price_usd: float,
    epoch_days: float,
) -> float:
    """Return the Q_me at which R_uncapped first equals cap_budget.

    Returns math.inf when cap is algebraically unreachable (denominator <= 0).
    """
    if pendle_price_usd <= 0.0:
        raise ValueError(f"pendle_price_usd must be > 0, got {pendle_price_usd}")
    if epoch_days <= 0.0:
        raise ValueError(f"epoch_days must be > 0, got {epoch_days}")

    cap_budget = 10.0 * collateral_usd * (epoch_days / 365.0) / pendle_price_usd

    numerator = sigma_others_pendle * cap_budget
    denominator = e_liq_per_epoch_pendle * side_split * t_frac - cap_budget * t_frac

    if denominator <= 0.0:
        # Cap is at or above E_liq × side_split × t_frac — unreachable at any finite Q_me
        return math.inf

    return numerator / denominator
```

"""REWARD-02: CompetitionTracker — 5s REST poll + 1 Hz parquet read.

Composes two sources:
  - Q_me via POST /open-api/v2/accounts/all-limit-orders (operator's own orders)
  - Σ_others via DuckDB aggregation over parquet.order_book_combined minus AMM share

AMM-share exclusion per Phase-2 fill_model.amm_consumption.compute_amm_share_timeseries
reuse — T-03-06 mitigation. AMMs don't compete for LO Liquidity Rewards.

Degraded-mode per RESEARCH §REWARD-02:
  - 5 consecutive REST failures → data_source='fallback_parquet_only', Q_me estimated from
    SQLite open_orders (Phase-1 writer), competition_tracker.rest_degraded event emitted.
"""
from __future__ import annotations

import asyncio
import sqlite3
from datetime import UTC, datetime
from pathlib import Path
from typing import Any

import structlog

from ..config.schema import BotConfig
from ..data.chain_time import now_cached as get_chain_time_cached
from ..state.writers.competition_history import insert_competition_row

log = structlog.get_logger(__name__)

_BACKOFF_DELAYS = [0.5, 1.0, 2.0, 4.0, 8.0]

__all__ = ["CompetitionTracker", "run_competition_tracker_loop"]


class CompetitionTracker:
    """Polls /open-api/v2/accounts/all-limit-orders + reads parquet.order_book_combined."""

    def __init__(
        self,
        config: BotConfig,
        conn: sqlite3.Connection,
        rest: Any,
        run_id: str,
        event_pub: Any,
        sqlite_path: Path,
    ) -> None:
        self.config = config
        self.conn = conn
        self.rest = rest
        self.run_id = run_id
        self.event_pub = event_pub
        self.sqlite_path = sqlite_path
        self.interval = config.reward_share_model.competition_poll_interval_s
        self.fail_threshold = config.reward_share_model.fallback_to_parquet_after_rest_fails_n
        self.markets = [m.market_id for m in config.markets]
        self._fail_count: int = 0

    async def run_one_cycle(self) -> None:
        """One (REST + parquet) pass; writes one competition_history row per market."""
        chain_ts = int(await get_chain_time_cached())
        polled_at = datetime.now(UTC).isoformat()

        for market_id in self.markets:
            try:
                q_me = await self._get_q_me_with_backoff(market_id)
                data_source = "live"
                self._fail_count = 0
            except Exception as exc:  # noqa: BLE001
                log.error(
                    "competition_tracker.rest_fetch_failed",
                    error=str(exc),
                    fail_count=self._fail_count,
                    market_id=market_id,
                )
                if self._fail_count >= self.fail_threshold:
                    await self.event_pub.publish(
                        "competition_tracker.rest_degraded",
                        {"fail_count": self._fail_count, "market_id": market_id},
                    )
                # Fallback: estimate Q_me from SQLite open_orders
                q_me = self._estimate_q_me_from_open_orders(market_id)
                data_source = "fallback_parquet_only"

            # Read Σ_others from parquet.order_book_combined (AMM-share excluded)
            sigma_others = self._read_sigma_others_from_parquet(market_id, chain_ts, q_me)

            insert_competition_row(
                self.conn,
                polled_at_utc=polled_at,
                chain_ts=chain_ts,
                market_id=market_id,
                q_me_pendle=q_me,
                sigma_others_pendle=sigma_others,
                run_id=self.run_id,
                data_source=data_source,
            )
            self.conn.commit()  # tracker owns its transaction lifecycle

    async def _get_q_me_with_backoff(self, market_id: str) -> float:
        """POST /open-api/v2/accounts/all-limit-orders — retry with backoff, return q_me in PENDLE-units."""
        last_exc: BaseException | None = None
        for delay in _BACKOFF_DELAYS:
            try:
                results = await self.rest.post_all_limit_orders(
                    user_address=self.config.root_wallet_address,
                    account_id=0,
                )
                # Filter to market + sum sizes in-range (clo_lower <= rate <= clo_upper)
                return self._sum_q_me(results, market_id)
            except Exception as exc:  # noqa: BLE001
                last_exc = exc
                self._fail_count += 1
                await asyncio.sleep(delay)
        raise last_exc  # type: ignore[misc]

    def _sum_q_me(self, orders: list[dict], market_id: str) -> float:
        """Sum our own order sizes for the market, in PENDLE-units. Operator-provided notional_per_yu applies."""
        # v0.5: assume REST returns sizes already in PENDLE-equivalent units via the wallet's accounting.
        # If the REST field is in YU, caller converts via MarketCalibration.notional_per_yu_usd / pendle_price_usd.
        # For now keep the scaling minimal; Plan 03-03 wires the full sizer math.
        total = 0.0
        for o in orders:
            if o.get("marketId") == market_id or o.get("symbol") == market_id:
                # Convert from YU wei -> float PENDLE proxy (1 YU = 1 PENDLE-equivalent in v0.5 placeholder)
                size_yu = float(o.get("sz", 0)) / 1e18 if isinstance(o.get("sz"), str) else float(o.get("sz", 0))
                total += size_yu
        return total

    def _read_sigma_others_from_parquet(self, market_id: str, chain_ts: int, q_me: float) -> float:
        """DuckDB query: sum raw_book size in [clo_lower, clo_upper] - q_me - AMM_share."""
        from ..duckdb_layer.setup import open_analytical_conn

        conn = open_analytical_conn(Path(self.config.infra.sqlite_path), Path("data/parquet/"))
        try:
            # Get latest clo bounds from state.clo_history (via attached SQLite in analytical_conn)
            clo_row = conn.execute(
                "SELECT clo_lower_thresh, clo_upper_thresh FROM clo_history "
                "WHERE market_id = ? ORDER BY snapshot_ts DESC LIMIT 1",
                [market_id],
            ).fetchone()
            if clo_row is None:
                log.warning("competition_tracker.no_clo_history", market_id=market_id)
                return 0.0
            clo_lower, clo_upper = clo_row

            # Sum raw_book (combined_book - amm_share) in the CLO range
            # Phase 1 writes parquet.order_book_combined with columns including `tick`, `sz_combined_float`, and
            # paired parquet.order_book_raw with `sz_raw_float`. AMM share = combined - raw per tick.
            # Sigma_others = raw book in range - q_me (we exclude AMM share by using RAW not COMBINED).
            # (This is RESEARCH §REWARD-02 AMM exclusion: AMMs are combined-raw delta; we read raw only.)
            sigma_df = conn.execute(
                """
                SELECT COALESCE(SUM(sz_float), 0.0) AS sum_sz
                FROM read_parquet('data/parquet/order_book_raw/market_id=' || ? || '/**/*.parquet')
                WHERE tick BETWEEN ? AND ?
                  AND snapshot_ts = (
                    SELECT MAX(snapshot_ts) FROM read_parquet('data/parquet/order_book_raw/market_id=' || ? || '/**/*.parquet')
                    WHERE snapshot_ts <= ?
                  )
                """,
                [market_id, clo_lower, clo_upper, market_id, chain_ts],
            ).pl()
            if sigma_df.height == 0:
                log.warning("competition_tracker.no_order_book_snapshot", market_id=market_id)
                return 0.0
            total_raw = float(sigma_df.item(0, 0))
        finally:
            conn.close()

        sigma_others = max(0.0, total_raw - q_me)
        return sigma_others

    def _estimate_q_me_from_open_orders(self, market_id: str) -> float:
        """Fallback when REST is degraded: sum sizes from SQLite state.open_orders (Phase-1 writer)."""
        cursor = self.conn.execute(
            "SELECT COALESCE(SUM(size_float), 0.0) FROM open_orders "
            "WHERE market_id = ? AND status IN ('filling','pending')",
            (market_id,),
        )
        row = cursor.fetchone()
        return float(row[0]) if row else 0.0


async def run_competition_tracker_loop(
    config: BotConfig,
    run_id: str,
    *,
    conn: sqlite3.Connection,
    rest: Any,
    event_pub: Any,
) -> None:
    """Spawned from brain.main via asyncio.create_task. Plan 03-04 wires this."""
    sqlite_path = Path(config.infra.sqlite_path)
    tracker = CompetitionTracker(config, conn, rest, run_id, event_pub, sqlite_path)
    log.info(
        "competition_tracker.loop_started",
        run_id=run_id,
        interval_s=tracker.interval,
    )
    while True:
        try:
            await tracker.run_one_cycle()
        except Exception as exc:  # noqa: BLE001
            log.exception("competition_tracker.cycle_failed", run_id=run_id, err=str(exc))
        await asyncio.sleep(tracker.interval)

# Reward cap geometry (v0.5)

**Status:** Current — v0.5 regime
**Filed:** 2026-04-24 (Phase 3, Plan 03-02)

## TL;DR
Piecewise `R_capped(Q_me) = min(R_uncapped(Q_me), cap_budget)` curve with three regimes:
sub_cap (Q_me small), cap_binding (R_uncapped ≥ cap), market_saturated (Σ_others=0).
At $100 collateral the cap_binding branch never fires on the ETHUSDT-29MAY2026 anchor.

## Formula
```
R_uncapped(Q_me) = E_liq_per_epoch × side_split × (Q_me × t_frac) / (Σ_others + Q_me × t_frac)
cap_budget_per_epoch_pendle = 10 × collateral_usd × (epoch_days / 365) / pendle_price_usd
R_capped(Q_me) = min(R_uncapped, cap_budget)
```

## Regime classifier
Returned as `RewardCurveOutput.regime: Literal["sub_cap","cap_binding","market_saturated"]`.
- `market_saturated` if `Σ_others == 0` and `Q_me > 0` — we own the market; full share.
- `cap_binding` if `R_uncapped >= cap_budget` — flat-top on the curve.
- `sub_cap` otherwise — healthy regime; at $100 this is always-true for the ETHUSDT anchor.

## Kinked-curve geometry (PITFALLS §A4 absorption)
The kink is at `Q_me` where `R_uncapped = cap_budget`. Below → slope > 0 (more size = more reward). Above → slope = 0 (cap clips). Q_cap solves algebraically (see [[empirical-loss-grid-v05]] sibling or test_reward_curve.py test_q_cap_finite_at_100_collateral_ethusdt).

- `side_split = 0.86` (short)
