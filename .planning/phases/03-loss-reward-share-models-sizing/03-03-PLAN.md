```python
"""REWARD-03 sync sizing optimizer. Phase 4 DecisionEngine calls this at 1 Hz.

Contract:
    - SYNC (not async). Composes fill_model.predict + loss_model.expected_loss +
      reward_model.r_capped_curve + reward_model.compute_q_cap.
    - p95 < 10ms budget (sub-call p95's ≤ 3ms each; optimizer overhead ≤ 5ms for 50-pt grid).
    - Propagates upstream exceptions without swallowing.
    - At $100 collateral: Q_me_target == Q_min trivially (collateral-constrained sub_cap).

Design-locked. Phase 4 DecisionEngine depends on this signature.
"""
from __future__ import annotations
import math
from pathlib import Path

import numpy as np
import scipy.optimize
import structlog

from ..config.schema import BotConfig
from ..fill_model.predict import predict
from ..fill_model.schemas import PredictInput
from ..loss_model.expected_loss import expected_loss
from ..loss_model.schemas import ExpectedLossInput
from ..reward_model.q_cap import compute_q_cap
from ..reward_model.r_capped import r_capped_curve
from ..reward_model.schemas import RewardCurveInput
from .schemas import InsufficientCollateralError, SizerInput, SizerOutput

log = structlog.get_logger(__name__)
__all__ = ["q_me_target"]


def q_me_target(
    req: SizerInput,
    *,
    sqlite_path: Path,
    config: BotConfig,
) -> SizerOutput:
    """Returns SizerOutput with q_me_target_pendle + regime + intermediate values."""
    # --- Step 1: Q_min / Q_max ---
    q_min = config.sizer.q_min_pendle
    # Q_max conversion: collateral_usd × safety_margin / notional_per_yu_usd → PENDLE
    q_max = (config.sizer.collateral_usd * config.sizer.q_max_safety_margin) / config.sizer.notional_per_yu_usd
    if q_min > q_max:
        raise InsufficientCollateralError(q_min, q_max)

    # --- Step 2: Q_cap (REWARD-04) ---
    q_cap = compute_q_cap(
        sigma_others_pendle=req.sigma_others_pendle,
        e_liq_per_epoch_pendle=req.e_liq_per_epoch_pendle,
        side_split=req.side_split,
        t_frac=req.t_frac,
        collateral_usd=config.sizer.collateral_usd,
        pendle_price_usd=config.pricing.pendle_price_usd,
        epoch_days=config.reward_share_model.epoch_days,
    )

    # --- Step 3: Pre-compute p_fill + e_loss (neither depends on Q_me) ---
    p_fill_output = predict(
        PredictInput(
            market_id=req.market_id,
            side=req.side,
            depth_from_mark_ticks=req.depth_from_mark_ticks,
            queue_ahead_wei=req.queue_ahead_wei,
            current_features=req.current_features,
            time_to_epoch_end_s=req.time_to_epoch_end_s,
            current_chain_ts=req.current_chain_ts,
        ),
        sqlite_path=sqlite_path,
        config=config,
    )
    e_loss_output = expected_loss(
        ExpectedLossInput(
            market_id=req.market_id,
            side=req.side,
            depth_from_mark_ticks=req.depth_from_mark_ticks,
            time_to_maturity_s=req.time_to_maturity_s,
            current_chain_ts=req.current_chain_ts,
        ),
        sqlite_path=sqlite_path,
        config=config,
    )
    p_fill = p_fill_output.p_fill_this_epoch
    e_loss = e_loss_output.expected_loss_pendle

    # --- Step 4: market_saturated shortcut ---
    if q_cap < q_min:
        # Cap binds even at minimum size → market_saturated
        r_out = r_capped_curve(
            RewardCurveInput(
                market_id=req.market_id,
                side=req.side,
                q_me_pendle=q_min,
                sigma_others_pendle=req.sigma_others_pendle,
                e_liq_per_epoch_pendle=req.e_liq_per_epoch_pendle,
                side_split=req.side_split,
                t_frac=req.t_frac,
                collateral_usd=config.sizer.collateral_usd,
                pendle_price_usd=config.pricing.pendle_price_usd,
                epoch_days=config.reward_share_model.epoch_days,
            )
        )
        ev = r_out.r_capped_pendle - e_loss * p_fill * q_min
        log.info("sizer.market_saturated_shortcut", market_id=req.market_id, side=req.side, q_cap=q_cap, q_min=q_min)
        return SizerOutput(
            q_me_target_pendle=q_min,
            regime="market_saturated",
            r_capped_at_target=r_out.r_capped_pendle,
            expected_loss_at_target=e_loss,
            p_fill_at_target=p_fill,
            ev_at_target=ev,
            q_min_pendle=q_min,
            q_max_pendle=q_max,
            q_cap_pendle=q_cap,
        )

    upper = min(q_max, q_cap) if not math.isinf(q_cap) else q_max

    # --- Step 5: Degenerate-interval shortcut (q_min == upper — collateral-constrained at $100) ---
    if upper <= q_min * 1.0001:
        r_out = r_capped_curve(
            RewardCurveInput(
                market_id=req.market_id, side=req.side, q_me_pendle=q_min,
                sigma_others_pendle=req.sigma_others_pendle,
                e_liq_per_epoch_pendle=req.e_liq_per_epoch_pendle,
                side_split=req.side_split, t_frac=req.t_frac,
                collateral_usd=config.sizer.collateral_usd,
                pendle_price_usd=config.pricing.pendle_price_usd,
                epoch_days=config.reward_share_model.epoch_days,
            )
        )
        ev = r_out.r_capped_pendle - e_loss * p_fill * q_min
        log.info("sizer.q_min_fixed_point", market_id=req.market_id, side=req.side, q_min=q_min, upper=upper)
        return SizerOutput(
            q_me_target_pendle=q_min,
            regime=r_out.regime,
            r_capped_at_target=r_out.r_capped_pendle,
            expected_loss_at_target=e_loss,
            p_fill_at_target=p_fill,
            ev_at_target=ev,
            q_min_pendle=q_min,
            q_max_pendle=q_max,
            q_cap_pendle=q_cap,
        )

    # --- Step 6: Grid search + optional scipy polish ---
    def _objective(q: float) -> float:
        r_out = r_capped_curve(
            RewardCurveInput(
                market_id=req.market_id, side=req.side, q_me_pendle=q,
                sigma_others_pendle=req.sigma_others_pendle,
                e_liq_per_epoch_pendle=req.e_liq_per_epoch_pendle,
                side_split=req.side_split, t_frac=req.t_frac,
                collateral_usd=config.sizer.collateral_usd,
                pendle_price_usd=config.pricing.pendle_price_usd,
                epoch_days=config.reward_share_model.epoch_days,
            )
        )
        return float(r_out.r_capped_pendle - e_loss * p_fill * q)

    q_grid = np.logspace(np.log10(q_min), np.log10(upper), config.sizer.optimizer_grid_points)
    objectives = np.array([_objective(float(q)) for q in q_grid])

    # T-03-10: NaN guard
    if np.any(np.isnan(objectives)):
        log.warning("sizer.degraded_scipy_input", market_id=req.market_id, side=req.side)
        # Fall back to grid-only
        q_opt = float(q_grid[int(np.nanargmax(objectives))])
    elif config.sizer.optimizer_method == "grid_only":
        q_opt = float(q_grid[int(np.argmax(objectives))])
    elif config.sizer.optimizer_method == "scipy_only":
        res = scipy.optimize.minimize_scalar(
            lambda q: -_objective(float(q)),
            bounds=(q_min, upper),
            method="bounded",
        )
        q_opt = float(res.x)
    else:  # grid_plus_scipy
        q_grid_best = float(q_grid[int(np.argmax(objectives))])
        # Polish in a narrow bracket around the grid argmax
        bracket_lo = max(q_min, q_grid_best * 0.5)
        bracket_hi = min(upper, q_grid_best * 2.0)
        try:
            res = scipy.optimize.minimize_scalar(
                lambda q: -_objective(float(q)),
                bounds=(bracket_lo, bracket_hi),
                method="bounded",
            )
            q_opt = float(res.x)
            # Guard: if scipy diverged outside grid sanity, revert
            if _objective(q_opt) < _objective(q_grid_best) - 1e-12:
                q_opt = q_grid_best
        except Exception as exc:  # noqa: BLE001
            log.warning("sizer.scipy_polish_failed", error=str(exc), fallback="grid")
            q_opt = q_grid_best

    # --- Step 7: Build SizerOutput at q_opt ---
    r_out_final = r_capped_curve(
        RewardCurveInput(
            market_id=req.market_id, side=req.side, q_me_pendle=q_opt,
            sigma_others_pendle=req.sigma_others_pendle,
            e_liq_per_epoch_pendle=req.e_liq_per_epoch_pendle,
            side_split=req.side_split, t_frac=req.t_frac,
            collateral_usd=config.sizer.collateral_usd,
            pendle_price_usd=config.pricing.pendle_price_usd,
            epoch_days=config.reward_share_model.epoch_days,
        )
    )
    ev_final = _objective(q_opt)

    log.info(
        "sizer.q_me_target_computed",
        market_id=req.market_id, side=req.side,
        q_me_target=q_opt, regime=r_out_final.regime, ev=ev_final,
        q_min=q_min, q_max=q_max, q_cap=q_cap,
    )
    return SizerOutput(
        q_me_target_pendle=q_opt,
        regime=r_out_final.regime,
        r_capped_at_target=r_out_final.r_capped_pendle,
        expected_loss_at_target=e_loss,
        p_fill_at_target=p_fill,
        ev_at_target=ev_final,
        q_min_pendle=q_min,
        q_max_pendle=q_max,
        q_cap_pendle=q_cap,
    )
```

# Sizing optimizer — sub-cap regime (v0.5)

**Status:** Current — v0.5 regime
**Filed:** 2026-04-24 (Phase 3, Plan 03-03)

## TL;DR
`brain.sizer.optimizer.q_me_target(req)` returns `Q_min` deterministically at $100 collateral because the collateral constraint collapses `Q_max → Q_min`. ## Objective
```
maximize: R_capped(Q_me) - E[loss|fill] × P(fill) × Q_me
subject to: Q_min ≤ Q_me ≤ min(Q_max, Q_cap)
```

- `E[loss|fill]` from [[empirical-loss-grid-v05]] (LOSS-02)
- `P(fill)` from [[fill-model-predict-contract]] (Phase 2 FILL-05)
- `Q_cap` from [[reward-cap-geometry-v05]] (REWARD-04 algebra)

## $100 sub-cap fixed point
With `config.sizer`:

## $5000 scale-up proof
With `collateral_usd = 5000.0`, `notional_per_yu_usd = 1.0`:
→ `Q_max = 5000 × 0.9 / 1 = 4500 PENDLE` (large).
→ `Q_cap ≈ 3.72 PENDLE` at the ETHUSDT anchor (see [[reward-cap-geometry-v05]]).
→ `upper = min(4500, 3.72) = 3.72`.
→ Grid search over `logspace(0.001, 3.72, 50)`; scipy polishes. Result: interior `Q_me_target < Q_cap` in cap_binding regime.

Test `test_sizer.py::test_scale_up_to_5k_cap_binding` asserts this.

## Algorithm
1. `q_min = config.sizer.q_min_pendle`.
2. `q_max = collateral_usd × q_max_safety_margin / notional_per_yu_usd`. If `q_min > q_max` → raise `InsufficientCollateralError`.
3. `q_cap = compute_q_cap(...)` (REWARD-04). If `q_cap < q_min` → shortcut: `regime='market_saturated'`, return `Q_me_target = q_min`.
4. `upper = min(q_max, q_cap)`. If `upper ≤ q_min × 1.0001` → degenerate-interval shortcut: return `Q_me_target = q_min` (this is the $100 sub-cap fixed point).
5. Pre-compute `p_fill` + `e_loss` (neither depends on Q_me).
6. Grid-search 50 log-spaced points.
7. Optional scipy polish in a bracket `[q_grid_best × 0.5, q_grid_best × 2.0]` (bounded minimize_scalar).

## Why pre-compute p_fill + e_loss outside the grid loop
Both depend on `depth_from_mark_ticks + time_to_maturity_s`, NOT on Q_me. Looping them inside the grid would waste 50× SQLite reads per Sizer call. Net p95 latency stays under 10ms.
