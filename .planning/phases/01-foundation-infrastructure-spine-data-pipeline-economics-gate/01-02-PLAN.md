    """SQLite WAL connection helpers. Single-writer-per-table discipline is enforced at
    the application layer by the lint in scripts/sqlite_writer_lint.py — this module just
    opens WAL connections and enforces chmod 600."""
    from __future__ import annotations
    import os, sqlite3
    from pathlib import Path
    from contextlib import contextmanager

    def ensure_permissions_0600(path: Path) -> None:
        """Set 0o600 on the SQLite file (and -wal/-shm siblings if present).
        Per RESEARCH §Security Domain — SQLite must not be world-readable."""
        for suffix in ("", "-wal", "-shm"):
            p = Path(str(path) + suffix)
            if p.exists():
                os.chmod(p, 0o600)

    def enable_wal_mode(conn: sqlite3.Connection) -> None:
        conn.execute("PRAGMA journal_mode = WAL")
        conn.execute("PRAGMA synchronous = NORMAL")  # WAL + NORMAL is the durable+fast combo
        conn.execute("PRAGMA foreign_keys = ON")

    @contextmanager
    def open_connection(path: Path | str):
        """Open a WAL-mode SQLite connection with FKs on. Creates parent dirs. 0o600 perms."""
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)
        conn = sqlite3.connect(str(path), isolation_level=None)  # autocommit; explicit txns use BEGIN
        enable_wal_mode(conn)
        ensure_permissions_0600(path)
        try:
            yield conn
        finally:
            conn.close()
    ```

    #!/usr/bin/env python3
    """Single-writer-per-table lint (INFRA-03, PROTO-02).

    For each of the 8 tables, verify that `INSERT INTO <table>` and `UPDATE <table> SET`
    appear ONLY in the designated owner file. Exit 0 on clean, exit 1 with violation list.

    Owner map (authoritative):
      open_orders           -> ts/src/boros_hands/state/sqlite_writer.ts
      tx_receipts           -> ts/src/boros_hands/state/sqlite_writer.ts
      decisions             -> python/src/boros_brain/state/writers/decisions.py
      queue_positions       -> python/src/boros_brain/state/writers/queue_positions.py
      run_manifests         -> python/src/boros_brain/state/writers/run_manifests.py
      epoch_rewards         -> python/src/boros_brain/state/writers/epoch_rewards.py
      clo_history           -> python/src/boros_brain/state/writers/clo_history.py
      reward_budget_history -> python/src/boros_brain/state/writers/reward_budget_history.py

    Note: Python writer modules for hands-owned tables (open_orders, tx_receipts) MAY
    contain INSERT (TEST_ONLY guard) — recognised by the presence of `TEST_ONLY = True`
    at module scope. The lint treats those files as ALLOWED writers.
    """
    import argparse, re, sys
    from pathlib import Path

    OWNER_MAP = {
        "open_orders":           "ts/src/boros_hands/state/sqlite_writer.ts",
        "tx_receipts":           "ts/src/boros_hands/state/sqlite_writer.ts",
        "decisions":             "python/src/boros_brain/state/writers/decisions.py",
        "queue_positions":       "python/src/boros_brain/state/writers/queue_positions.py",
        "run_manifests":         "python/src/boros_brain/state/writers/run_manifests.py",
        "epoch_rewards":         "python/src/boros_brain/state/writers/epoch_rewards.py",
        "clo_history":           "python/src/boros_brain/state/writers/clo_history.py",
        "reward_budget_history": "python/src/boros_brain/state/writers/reward_budget_history.py",
    }
    TEST_ONLY_EXCEPTION_FILES = {
        "open_orders":  "python/src/boros_brain/state/writers/open_orders.py",
        "tx_receipts":  "python/src/boros_brain/state/writers/tx_receipts.py",
    }

    def scan(root: Path) -> list[str]:
        violations = []
        all_src = list((root / "python" / "src").rglob("*.py")) + list((root / "ts" / "src").rglob("*.ts"))
        for table, owner_rel in OWNER_MAP.items():
            pat = re.compile(rf"(INSERT INTO {table}\b|UPDATE {table}\b|ON CONFLICT.*{table}\b)", re.IGNORECASE)
            for src in all_src:
                rel = src.relative_to(root).as_posix()
                if rel == owner_rel:
                    continue
                # TEST_ONLY exception for hands-owned tables
                if table in TEST_ONLY_EXCEPTION_FILES and rel == TEST_ONLY_EXCEPTION_FILES[table]:
                    txt = src.read_text()
                    if "TEST_ONLY = True" in txt:
                        continue
                txt = src.read_text()
                if pat.search(txt):
                    violations.append(f"{rel}: writes to `{table}` (owner: {owner_rel})")
        return violations

    def main() -> int:
        ap = argparse.ArgumentParser()
        ap.add_argument("--root", default=".", type=Path)
        args = ap.parse_args()
        vs = scan(args.root)
        if vs:
            print("\n".join(vs), file=sys.stderr)
            return 1
        return 0

    if __name__ == "__main__":
        sys.exit(main())
    ```
