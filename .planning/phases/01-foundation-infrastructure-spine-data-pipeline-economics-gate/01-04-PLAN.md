    ```python
    #!/usr/bin/env python3
    """Architecture drift lint (INFRA-09, PROTO-02).

    Walks python/src/ via ast + ts/src/ via regex; diffs against docs/architecture.yml.
    Exit 0 on clean, exit 1 with violation list on drift.

    Granularity (per RESEARCH §Pattern 5 granularity rule):
      - Declare: modules + public top-level functions + CLI commands + config knobs + external deps
      - Do NOT declare: private helpers, dataclass fields, methods on non-public classes
    """
    from __future__ import annotations
    import argparse, ast, re, sys
    from pathlib import Path
    import yaml

    def extract_python_inventory(src_root: Path) -> dict:
        modules: set[str] = set()
        functions: set[tuple[str, str]] = set()  # (module_id, function_name)
        for py in src_root.rglob("*.py"):
            if py.name.startswith("_") and py.name != "__init__.py":
                continue
            # module_id: e.g. python/src/boros_brain/config/schema.py -> brain.config.schema
            rel_parts = py.relative_to(src_root).with_suffix("").parts
            if rel_parts[-1] == "__init__":
                rel_parts = rel_parts[:-1]
            if not rel_parts:
                continue
            # drop the package root (e.g. boros_brain) and treat it as the process id
            process_name = "brain" if rel_parts[0] == "boros_brain" else "hands"
            mod_id = ".".join([process_name] + list(rel_parts[1:]))
            if not rel_parts[1:]:
                continue  # top-level __init__ handled separately
            modules.add(mod_id)
            try:
                tree = ast.parse(py.read_text())
            except SyntaxError:
                continue
            for node in tree.body:
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and not node.name.startswith("_"):
                    functions.add((mod_id, node.name))
        return {"modules": modules, "functions": functions}

    def extract_ts_inventory(src_root: Path) -> dict:
        modules: set[str] = set()
        for ts in src_root.rglob("*.ts"):
            if ts.name.endswith(".test.ts"):
                continue
            rel_parts = ts.relative_to(src_root).with_suffix("").parts
            process_name = "hands" if rel_parts[0] == "boros_hands" else "brain"
            mod_id = ".".join([process_name] + list(rel_parts[1:]))
            if not rel_parts[1:]:
                continue
            modules.add(mod_id)
        return {"modules": modules}

    def compare(yml: dict, py_inv: dict, ts_inv: dict) -> list[str]:
        violations: list[str] = []
        declared_modules = {n["id"] for n in yml.get("nodes", []) if n.get("type") == "module"}
        code_modules = py_inv["modules"] | ts_inv["modules"]

        undeclared = code_modules - declared_modules
        orphaned = declared_modules - code_modules
        for m in sorted(undeclared):
            violations.append(f"Code defines module `{m}` not declared in architecture.yml")
        for m in sorted(orphaned):
            violations.append(f"architecture.yml declares module `{m}` not found in source tree")

        # Verify declared paths actually exist
        for n in yml.get("nodes", []):
            if n.get("type") != "module":
                continue
            p = n.get("path")
            if p and not Path(p).exists():
                violations.append(f"architecture.yml node `{n['id']}` points at missing path `{p}`")

        # Verify declared public_functions exist in the source
        for n in yml.get("nodes", []):
            if n.get("type") != "module":
                continue
            declared_fns = set(n.get("public_functions") or [])
            actual_fns = {fn for (mod, fn) in py_inv["functions"] if mod == n["id"]}
            for fn in declared_fns - actual_fns:
                # Only Python module fns are checked (TS discovery is regex-lite)
                if n.get("path", "").endswith(".py"):
                    violations.append(f"architecture.yml declares public fn `{n['id']}.{fn}` not found in code")
        return violations

    def main() -> int:
        ap = argparse.ArgumentParser()
        ap.add_argument("--yml", default="docs/architecture.yml", type=Path)
        ap.add_argument("--src", default="python/src", type=Path)
        ap.add_argument("--ts-src", default="ts/src", type=Path)
        args = ap.parse_args()
        yml = yaml.safe_load(args.yml.read_text())
        py_inv = extract_python_inventory(args.src) if args.src.exists() else {"modules": set(), "functions": set()}
        ts_inv = extract_ts_inventory(args.ts_src) if args.ts_src.exists() else {"modules": set()}
        vs = compare(yml, py_inv, ts_inv)
        if vs:
            print("\n".join(vs), file=sys.stderr)
            print(f"\n{len(vs)} architecture-lint violation(s) — update docs/architecture.yml", file=sys.stderr)
            return 1
        return 0

    if __name__ == "__main__":
        sys.exit(main())
    ```

    ```python
    #!/usr/bin/env python3
    """Wiki-cite lint (INFRA-11, PROTO-03).

    Any new markdown under research/, decisions/, postmortem/ (CLAUDE.md Engineering Protocol #1
    write-back rule) must contain at least one `[[wiki-link]]` reference OUTSIDE of code blocks.

    Exit 0 on clean, 1 on violation.

    Usage:
      scripts/wiki_cite_lint.py <path>...   # lint specific files (from pre-commit hook)
      scripts/wiki_cite_lint.py --all       # lint all files under the policed directories
    """
    from __future__ import annotations
    import argparse, re, sys
    from pathlib import Path

    POLICED_PATH_PARTS = ("research", "decisions", "postmortem")
    WIKI_LINK_RE = re.compile(r"\[\[([^\]\n]+?)\]\]")

    def strip_code_blocks(text: str) -> str:
        # Remove fenced code blocks (```...```) and inline code (`...`) before searching for [[link]]
        # This enforces cites appearing in rendered prose, not hidden in code examples.
        text = re.sub(r"```.*?```", "", text, flags=re.DOTALL)
        text = re.sub(r"`[^`\n]*`", "", text)
        return text

    def has_wiki_link(md_path: Path) -> bool:
        text = md_path.read_text()
        stripped = strip_code_blocks(text)
        return WIKI_LINK_RE.search(stripped) is not None

    def is_policed(path: Path) -> bool:
        # path may be absolute; look at any path component matching POLICED_PATH_PARTS
        return any(part in POLICED_PATH_PARTS for part in path.parts) and path.suffix == ".md"

    def main() -> int:
        ap = argparse.ArgumentParser()
        ap.add_argument("paths", nargs="*", type=Path)
        ap.add_argument("--all", action="store_true", help="Lint every .md under policed dirs")
        args = ap.parse_args()
        candidates: list[Path] = []
        if args.all:
            for part in POLICED_PATH_PARTS:
                for md in Path(".").rglob(f"{part}/**/*.md"):
                    candidates.append(md)
        else:
            candidates = [p for p in args.paths if is_policed(p)]

        violations: list[str] = []
        for md in candidates:
            if not has_wiki_link(md):
                violations.append(f"{md}: no [[wiki-link]] reference outside code blocks")
        if violations:
            print("\n".join(violations), file=sys.stderr)
            print(f"\n{len(violations)} wiki-cite violation(s) — file a wiki page in ../LLM\\ Wiki/workspaces/boros/wiki/ and cite it.", file=sys.stderr)
            return 1
        return 0

    if __name__ == "__main__":
        sys.exit(main())
    ```

    ```python
    #!/usr/bin/env python3
    """Plain-key lint (security, PROTO-05-adjacent).

    Greps staged files for patterns matching `0x<64 hex chars>` (Ethereum private key) and
    `-----BEGIN PRIVATE KEY-----` / `-----BEGIN EC PRIVATE KEY-----` / `-----BEGIN RSA PRIVATE KEY-----`.

    Exceptions: documentation examples using `FAKE_KEY_FOR_TESTS = "0x" + "0"*64` pattern
    are allowed (zero-hex detected separately).
    """
    import argparse, re, sys
    from pathlib import Path

    ETH_KEY_RE = re.compile(r"\b0x[a-fA-F0-9]{64}\b")
    ALL_ZERO_RE = re.compile(r"\b0x0{64}\b")
    PEM_RE = re.compile(r"-----BEGIN (?:RSA |EC )?PRIVATE KEY-----")

    def scan_file(path: Path) -> list[str]:
        try:
            text = path.read_text()
        except (UnicodeDecodeError, PermissionError):
            return []
        violations = []
        for m in ETH_KEY_RE.finditer(text):
            if ALL_ZERO_RE.match(m.group(0)):
                continue  # all-zero synthetic test key
            violations.append(f"{path}:{text[:m.start()].count(chr(10))+1}: 64-hex private-key pattern")
        if PEM_RE.search(text):
            violations.append(f"{path}: PEM private key header detected")
        return violations

    def main() -> int:
        ap = argparse.ArgumentParser()
        ap.add_argument("paths", nargs="*", type=Path)
        args = ap.parse_args()
        all_vs: list[str] = []
        for p in args.paths:
            if p.is_file():
                all_vs.extend(scan_file(p))
        if all_vs:
            print("\n".join(all_vs), file=sys.stderr)
            return 1
        return 0

    if __name__ == "__main__":
        sys.exit(main())
    ```
