    ```python
    """RUN_MANIFEST writer (INFRA-06, PROTO-01).

    Manifest is built by reflecting over BotConfig.model_dump(mode='json') — adding a knob
    to BotConfig without updating this file is impossible because the reflection walks
    every nested StrictBase recursively. PITFALLS §D2 prevention.
    """
    from __future__ import annotations
    import hashlib, json, subprocess, sqlite3
    from datetime import datetime, timezone
    from pathlib import Path
    from ulid import ULID
    from ..config.schema import BotConfig
    from ..state.writers.run_manifests import insert_run_manifest
    from .schema import RunManifest

    def _git_sha() -> str:
        try:
            return subprocess.check_output(['git', 'rev-parse', 'HEAD'], stderr=subprocess.DEVNULL).decode().strip()
        except subprocess.CalledProcessError:
            return "0" * 40  # unversioned checkout (CI sandbox etc.)

    def _git_dirty() -> bool:
        try:
            return subprocess.call(['git', 'diff', '--quiet'], stderr=subprocess.DEVNULL) != 0
        except FileNotFoundError:
            return False

    def _wallet_address_hash(addr: str) -> str:
        return hashlib.sha256(addr.lower().encode()).hexdigest()[:16]

    def build_run_manifest(config: BotConfig, parent_run_id: str | None = None) -> RunManifest:
        return RunManifest(
            run_id=str(ULID()),
            parent_run_id=parent_run_id,
            started_at=datetime.now(timezone.utc).isoformat(),
            mode=config.mode,
            strategy_version=config.strategy_version,
            git_sha=_git_sha(),
            git_dirty=_git_dirty(),
            market_id=config.market_id,
            side=config.side,
            wallet_address_hash=_wallet_address_hash(config.root_wallet_address),
            config=config.model_dump(mode='json'),
            chain_time_drift_at_start_seconds=None,
            agent_wallet_gas_balance_at_start=None,
        )

    def write_manifest(manifest: RunManifest, logs_dir: Path, sqlite_path: Path) -> Path:
        """Write sidecar JSON to logs/<YYYY-MM-DD>/<run_id>.manifest.json AND insert into run_manifests table."""
        date_dir = logs_dir / datetime.now(timezone.utc).strftime("%Y-%m-%d")
        date_dir.mkdir(parents=True, exist_ok=True)
        sidecar_path = date_dir / f"{manifest.run_id}.manifest.json"
        sidecar_path.write_text(json.dumps(manifest.model_dump(mode='json'), indent=2))
        from ..state.db import open_connection
        with open_connection(sqlite_path) as conn:
            insert_run_manifest(
                conn,
                run_id=manifest.run_id,
                parent_run_id=manifest.parent_run_id,
                started_at_iso=manifest.started_at,
                mode=manifest.mode,
                strategy_version=manifest.strategy_version,
                git_sha=manifest.git_sha,
                git_dirty=1 if manifest.git_dirty else 0,
                market_id=manifest.market_id,
                side=manifest.side,
                wallet_address_hash=manifest.wallet_address_hash,
                config_json=json.dumps(manifest.config),
            )
            conn.commit()
        return sidecar_path
    ```

    ```python
    """structlog configuration with contextvars-bound run_id (INFRA-07, PROTO-01).

    The `merge_contextvars` processor pulls run_id + any scoped keys into every log record.
    `bind_contextvars` at startup survives asyncio.create_task and asyncio.gather automatically
    (Python 3.7+ contextvars semantics) — verified in test_structlog_contextvars.
    """
    from __future__ import annotations
    import logging, os
    from datetime import datetime, timezone
    from pathlib import Path
    import orjson, structlog
    from structlog.contextvars import bind_contextvars

    def configure_logging(run_id: str, tier: str, logs_dir: Path, first_manifest_line: dict | None = None) -> None:
        date_str = datetime.now(timezone.utc).strftime("%Y-%m-%d")
        date_dir = logs_dir / date_str
        date_dir.mkdir(parents=True, exist_ok=True)
        log_path = date_dir / f"{run_id}.{tier}.log.jsonl"
        # Open in append, binary, unbuffered so each line flushes immediately
        log_file = open(log_path, "ab", buffering=0)
        structlog.configure(
            processors=[
                structlog.contextvars.merge_contextvars,
                structlog.processors.add_log_level,
                structlog.processors.TimeStamper(fmt="iso", utc=True, key="ts"),
                structlog.processors.CallsiteParameterAdder(
                    parameters=[
                        structlog.processors.CallsiteParameter.MODULE,
                        structlog.processors.CallsiteParameter.FUNC_NAME,
                        structlog.processors.CallsiteParameter.LINENO,
                        structlog.processors.CallsiteParameter.PROCESS,
                    ]
                ),
                structlog.processors.dict_tracebacks,
                structlog.processors.JSONRenderer(serializer=orjson.dumps),
            ],
            wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
            context_class=dict,
            logger_factory=structlog.WriteLoggerFactory(file=log_file),
            cache_logger_on_first_use=True,
        )
        bind_contextvars(run_id=run_id, tier=tier, pid=os.getpid())
        # MANIFEST-FIRST invariant: write RUN_MANIFEST as the first log line
        if first_manifest_line is not None:
            log = structlog.get_logger("boros_brain.manifest.writer")
            log.info("RUN_MANIFEST", **first_manifest_line)

    def get_logger(name: str):
        return structlog.get_logger(name)
    ```

    **`python/src/boros_brain/main.py`** (updated — replaces the 01-01 stub):
    ```python
    """Entrypoint for brain process. MANIFEST-FIRST protocol:
       1. Load config (01-01)
       2. Build RUN_MANIFEST (this plan)
       3. Configure structlog with run_id + emit RUN_MANIFEST as first log line + sidecar JSON
       4. (Future plans: start ZMQ bus, heartbeat, data pipeline, decision engine)
    """
    from __future__ import annotations
    import argparse, asyncio, signal, sys
    from pathlib import Path
    from .config.loader import load_config
    from .manifest.writer import build_run_manifest, write_manifest
    from .logging.structlog_setup import configure_logging, get_logger

    async def run(config_path: Path, dry_run: bool = False) -> int:
        cfg = load_config(config_path)
        manifest = build_run_manifest(cfg)
        logs_dir = Path(cfg.infra.logs_dir)
        sqlite_path = Path(cfg.infra.sqlite_path)
        write_manifest(manifest, logs_dir, sqlite_path)
        configure_logging(
            run_id=manifest.run_id,
            tier="brain",
            logs_dir=logs_dir,
            first_manifest_line=manifest.model_dump(mode='json'),
        )
        log = get_logger("boros_brain.main")
        log.info("brain_booted", mode=cfg.mode, run_id=manifest.run_id)
        if dry_run:
            log.info("dry_run_exit")
            return 0
        # Install SIGTERM handler for clean shutdown (launchd sends SIGTERM)
        loop = asyncio.get_running_loop()
        stop = asyncio.Event()
        for sig in (signal.SIGTERM, signal.SIGINT):
            loop.add_signal_handler(sig, stop.set)
        # Future plans (01-03 Task 2) wire ZMQ bus here
        await stop.wait()
        log.info("brain_shutdown")
        return 0

    def main() -> int:
        ap = argparse.ArgumentParser()
        ap.add_argument("--mode", choices=["live","paper","backtest","ingest","screen"], default="paper")
        ap.add_argument("--config", type=Path, default=Path("strategy/v0.5-default.yaml"))
        ap.add_argument("--dry-run", action="store_true", help="Emit manifest + exit 0")
        args = ap.parse_args()
        return asyncio.run(run(args.config, dry_run=args.dry_run))

    if __name__ == "__main__":
        sys.exit(main())
    ```

    ```python
    """5s heartbeat + 15s (3-miss) lost-sibling protocol (INFRA-05 per RESEARCH §Pattern 3)."""
    from __future__ import annotations
    import asyncio, time
    from .events import EventBus
    from .messages import Heartbeat, LostSibling

    LOST_SIBLING_EVENT = "lost_sibling"

    class HeartbeatMonitor:
        def __init__(self, events: EventBus, tier: str, run_id: str, interval_s: float = 5.0, timeout_misses: int = 3):
            self.events = events; self.tier = tier; self.run_id = run_id
            self.interval_s = interval_s
            self.timeout_s = interval_s * timeout_misses
            self.last_peer_heartbeat_ms = 0
            self._stopped = asyncio.Event()

        async def send_loop(self) -> None:
            while not self._stopped.is_set():
                await self.events.publish(Heartbeat(from_tier=self.tier, ts_ms=int(time.time()*1000), run_id=self.run_id))
                try:
                    await asyncio.wait_for(self._stopped.wait(), timeout=self.interval_s)
                except asyncio.TimeoutError:
                    pass

        async def monitor_loop(self) -> None:
            self.last_peer_heartbeat_ms = int(time.time()*1000)  # prime
            while not self._stopped.is_set():
                await asyncio.sleep(self.interval_s)
                now_ms = int(time.time()*1000)
                if now_ms - self.last_peer_heartbeat_ms > self.timeout_s * 1000:
                    observer = "brain" if self.tier == "brain" else "hands"
                    await self.events.publish(LostSibling(observer=observer, missing_since_ts_ms=self.last_peer_heartbeat_ms, run_id=self.run_id))
                    # Stay in LOST_SIBLING state; recovery = peer resumes heartbeats

        def record_peer_heartbeat(self, ts_ms: int) -> None:
            self.last_peer_heartbeat_ms = ts_ms

        def stop(self) -> None:
            self._stopped.set()
    ```

    ```typescript
    import { CmdSchema, KillCmdSchema, type KillCmd, type PlaceCmd, type CancelCmd } from './messages.js';
    import { log } from '../logger.js';

    export type CmdHandler = (cmd: PlaceCmd | CancelCmd) => Promise<void>;
    export type KillHandler = (cmd: KillCmd) => Promise<void>;

    export class CmdListener {
      private cmdSock = new Pull();
      private killSock = new Pull();
      private running = true;

      async connect(cmdEndpoint: string, killEndpoint: string): Promise<void> {
        this.cmdSock.receiveHighWaterMark = 1000;
        this.killSock.receiveHighWaterMark = 1000;
        await this.cmdSock.connect(cmdEndpoint);
        await this.killSock.connect(killEndpoint);
      }

      // Separate drain loops; kill drain runs as a prioritized independent task
      async drainKill(onKill: KillHandler): Promise<void> {
        for await (const [buf] of this.killSock) {
          if (!this.running) break;
          const msg = KillCmdSchema.parse(JSON.parse(buf.toString()));
          await onKill(msg);
        }
      }
      async drainCmd(onCmd: CmdHandler): Promise<void> {
        for await (const [buf] of this.cmdSock) {
          if (!this.running) break;
          const parsed = CmdSchema.parse(JSON.parse(buf.toString()));
          if (parsed.kind === 'kill') continue;  // should never arrive here; defense-in-depth
          await onCmd(parsed);
        }
      }
      stop(): void { this.running = false; this.cmdSock.close(); this.killSock.close(); }
    }
    ```
