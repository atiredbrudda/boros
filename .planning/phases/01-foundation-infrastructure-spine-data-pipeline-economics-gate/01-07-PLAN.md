```python
"""INFRA-14 — chain-time cache + drift monitor.

Startup: refuses to boot if |time.time() - chain_time| > config.data.chain_time_warn_drift_s.
Runtime: drift > warn logs WARN; drift > halt triggers the registered halt callback
(normally publishes HALT on the ZMQ KILL priority lane — see bus/cmd.py).

Anti-pattern (lint-enforced by scripts/no_time_time_lint.py):
    import time; time.time()  # in brain/{decision,data,screener}/  -- FORBIDDEN.
All chain-relevant comparisons use now_cached().
"""
from __future__ import annotations
import asyncio, time
from dataclasses import dataclass, field
from typing import Awaitable, Callable, Optional
import structlog
from web3 import AsyncWeb3
from boros_brain.config.schema import BotConfig
from boros_brain.data.rpc_client import get_latest_block_timestamp

log = structlog.get_logger(__name__)


class ChainTimeDriftHalt(RuntimeError):
    """Raised when drift exceeds startup threshold."""


HaltCallback = Callable[[str, dict], Awaitable[None]]


async def _default_halt_callback(reason: str, payload: dict) -> None:
    # Fallback when bus isn't wired yet — log only.
    log.error("chain_time.halt_callback_unwired", reason=reason, **payload)


@dataclass
class ChainTimeCache:
    w3: AsyncWeb3
    config: BotConfig
    halt_callback: HaltCallback = field(default=_default_halt_callback)
    _value: Optional[int] = None
    _ts: float = 0.0

    async def now_cached(self) -> int:
        """Return cached chain timestamp (unix seconds). TTL = config.data.chain_time_cache_ttl_s."""
        now_wall = time.monotonic()
        ttl = self.config.data.chain_time_cache_ttl_s
        if self._value is None or (now_wall - self._ts) >= ttl:
            try:
                self._value = await asyncio.wait_for(
                    get_latest_block_timestamp(self.w3), timeout=5.0
                )
                self._ts = now_wall
            except asyncio.TimeoutError:
                log.warning("chain_time.rpc_timeout", last_value=self._value, stale=True)
                # Return stale cached value — do NOT halt on transient timeout.
        return self._value  # type: ignore[return-value]

    async def startup_check(self) -> float:
        """Called by main.py at boot. Returns initial drift in seconds (for RUN_MANIFEST).
        Raises ChainTimeDriftHalt if |time.time() - chain_time| > warn_drift_s."""
        chain_ts = await get_latest_block_timestamp(self.w3)
        self._value, self._ts = chain_ts, time.monotonic()
        drift = time.time() - chain_ts
        thresh = self.config.data.chain_time_warn_drift_s
        if abs(drift) > thresh:
            raise ChainTimeDriftHalt(
                f"Chain-time drift at startup |{drift:+.2f}s| exceeds threshold {thresh}s. "
                "Check NTP; likely macOS sleep-wake state. Refusing to start."
            )
        log.info("chain_time.startup_ok", drift_s=round(drift, 3))
        return drift

    async def drift_monitor_tick(self) -> None:
        """Called ~every 60s by a supervisor task. Logs WARN / halts on thresholds."""
        chain_ts = await self.now_cached()
        drift = time.time() - chain_ts
        if abs(drift) > self.config.data.chain_time_halt_drift_s:
            await self.halt_callback(
                "chain_time_drift",
                {"drift_s": abs(drift), "threshold_s": self.config.data.chain_time_halt_drift_s},
            )
        elif abs(drift) > self.config.data.chain_time_warn_drift_s:
            log.warning("chain_time.drift_warn", drift_s=round(drift, 3))


# Convenience: the main.py wires a module-level singleton that `now_cached()`
# delegates to. Other hot-path modules import `now_cached` directly.

_GLOBAL_CACHE: Optional[ChainTimeCache] = None


def install(cache: ChainTimeCache) -> None:
    global _GLOBAL_CACHE
    _GLOBAL_CACHE = cache


async def now_cached() -> int:
    if _GLOBAL_CACHE is None:
        raise RuntimeError("chain_time not installed — call install(cache) at main.py startup")
    return await _GLOBAL_CACHE.now_cached()


async def startup_check() -> float:
    if _GLOBAL_CACHE is None:
        raise RuntimeError("chain_time not installed")
    return await _GLOBAL_CACHE.startup_check()


async def drift_monitor_tick() -> None:
    if _GLOBAL_CACHE is None:
        raise RuntimeError("chain_time not installed")
    await _GLOBAL_CACHE.drift_monitor_tick()
```

```python
#!/usr/bin/env python3
"""Pre-commit hook: rejects `time.time()` in brain/{decision,data,screener}/.
Per CLAUDE.md + PITFALLS §C7: chain-time is the only clock that matters; comparing to time.time() drifts on macOS sleep/wake.
"""
import re, subprocess, sys
from pathlib import Path

PROTECTED_DIRS = (
    "python/src/boros_brain/decision/",
    "python/src/boros_brain/data/",
    "python/src/boros_brain/screener/",
)
# Allow time.time() in chain_time.py ONLY (it's the module that explicitly measures drift vs chain_time).
ALLOWED_FILES = ("python/src/boros_brain/data/chain_time.py",)
PATTERN = re.compile(r"\btime\.time\s*\(")

def staged_files() -> list[Path]:
    try:
        out = subprocess.check_output(
            ["git", "diff", "--cached", "--name-only", "--diff-filter=ACMR"], text=True
        )
        files = [p.strip() for p in out.splitlines() if p.strip()]
    except subprocess.CalledProcessError:
        files = []
    if not files:
        # Also support `--all-files` invocation (run-all): fall back to full tree walk
        files = [
            str(p) for d in PROTECTED_DIRS for p in Path(d).rglob("*.py") if p.is_file()
        ]
    return [Path(f) for f in files]

def scan(path: Path) -> list[tuple[int, str]]:
    s = str(path).replace("\\", "/")
    if not any(s.startswith(d) for d in PROTECTED_DIRS):
        return []
    if s in ALLOWED_FILES:
        return []
    if not path.is_file():
        return []
    hits = []
    for i, line in enumerate(path.read_text().splitlines(), start=1):
        if PATTERN.search(line):
            hits.append((i, line))
    return hits

def main() -> int:
    fail = 0
    for p in staged_files():
        for lineno, line in scan(p):
            print(f"[no_time_time] {p}:{lineno}: use chain_time.now_cached() — {line.strip()[:80]}", file=sys.stderr)
            fail = 1
    return fail

if __name__ == "__main__":
    sys.exit(main())
```

```python
"""INFRA-04 — DuckDB read-only analytical layer over parquet + SQLite.

Usage:
    con = open_analytical_conn('var/state.sqlite', 'data/parquet')
    con.execute('SELECT * FROM clo_history').fetchall()            # from SQLite via sqlite_scanner
    con.execute('SELECT * FROM market_trades LIMIT 10').fetchall() # from parquet via Hive partition

NEVER call from brain/{decision,bus,manifest}/ — this is analytical-only.
"""
from __future__ import annotations
from pathlib import Path
import duckdb
import structlog

log = structlog.get_logger(__name__)

_VIEWS_SQL_PATH = Path(__file__).parent / "views.sql"
_PARQUET_SUBDIRS = (
    "market_trades", "order_book_raw", "order_book_combined",
    "underlying_apr", "market_data", "settlement",
)


def attach_sqlite_readonly(con: duckdb.DuckDBPyConnection, sqlite_path: str) -> None:
    """Attach the live SQLite spine read-only via sqlite_scanner extension."""
    con.install_extension("sqlite_scanner")
    con.load_extension("sqlite_scanner")
    # DuckDB syntax: ATTACH '<path>' AS <alias> (TYPE sqlite, READ_ONLY);
    # The 'main' alias surfaces SQLite tables as first-class in the catalog.
    con.execute(f"ATTACH '{sqlite_path}' AS live (TYPE sqlite, READ_ONLY)")
    con.execute("USE live")


def register_parquet_views(con: duckdb.DuckDBPyConnection, parquet_root: str) -> None:
    """Register Hive-partitioned parquet views. Subdirs with no .parquet files get an empty stub view."""
    ddl = _VIEWS_SQL_PATH.read_text()
    root = Path(parquet_root)
    for subdir in _PARQUET_SUBDIRS:
        subdir_path = root / subdir
        has_files = subdir_path.exists() and any(subdir_path.rglob("*.parquet"))
        if has_files:
            # Substitute and execute the matching CREATE VIEW line
            # simple line-based filter
            for stmt in ddl.split(";"):
                if subdir in stmt and "CREATE OR REPLACE VIEW" in stmt:
                    con.execute(stmt.format(parquet_root=str(root)))
        else:
            # Empty stub view — lets SELECT ... LIMIT 1 return 0 rows without error.
            # Schema will be widened by the real view once parquet exists.
            con.execute(
                f"CREATE OR REPLACE VIEW {subdir} AS SELECT NULL::VARCHAR AS _placeholder WHERE FALSE"
            )
            log.info("duckdb.empty_parquet_view_registered", subdir=subdir, path=str(subdir_path))


def open_analytical_conn(sqlite_path: str, parquet_root: str) -> duckdb.DuckDBPyConnection:
    """Returns a duckdb connection with SQLite attached read-only and parquet views registered.

    Memory connection — extensions need write. SQLite attach is READ_ONLY.
    parquet_root may be empty (plan 01-08 has not yet populated it); views are stubbed in that case.
    """
    con = duckdb.connect(":memory:", read_only=False)
    attach_sqlite_readonly(con, sqlite_path)
    register_parquet_views(con, parquet_root)
    return con
```
