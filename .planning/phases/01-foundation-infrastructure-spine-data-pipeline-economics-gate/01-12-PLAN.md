    ```python
    """DATA-09 economics kill-switch.

    Formula (verbatim per 01-RESEARCH.md §Pattern 10):
        R_apr_on_collateral = R_share_per_epoch × epochs_per_year / required_collateral
        R_share_per_epoch = E_liq_per_epoch × side_split × (Q_me × t_frac) / (Σ_others + Q_me × t_frac)
        epochs_per_year = 52 (PROJECT.md L63, weekly epochs)
        Kill: drop(market) iff R_apr_on_collateral < alt_yield + fill_loss_coarse_bound
    """
    import json
    from pathlib import Path
    from datetime import datetime, timezone
    from typing import Any
    from structlog import get_logger
    from ..config.loader import load_raw_yaml, load_config
    from ..config.schema import BotConfig
    from .kill_report import write_kill_report

    log = get_logger(__name__)

    EPOCHS_PER_YEAR = 52   # PROJECT.md L63 — weekly epochs (Fri 00:00 UTC → Thu 23:59 UTC). Locked constant.

    class ConfigError(RuntimeError): ...
    class ProjectHalt(SystemExit):
        """Signals Phase 1 economics-gate halt. CLI converts to exit code 1."""

    def _assert_knobs_explicit(raw_yaml: dict) -> None:
        """Per locked decision: kill-switch refuses silent defaults.
        Both knobs MUST be present as explicit keys under `screener:` in the YAML."""
        screener = raw_yaml.get("screener", {})
        if "alt_yield" not in screener:
            raise ConfigError("knob alt_yield missing — kill-switch refuses to run with silent default")
        if "fill_loss_coarse_bound" not in screener:
            raise ConfigError("knob fill_loss_coarse_bound missing — kill-switch refuses to run with silent default")

    def _assert_knobs_in_range(config: BotConfig) -> None:
        """Unit-confusion guard: alt_yield and fill_loss_coarse_bound are decimal fractions."""
        if not (0.0 < config.screener.alt_yield < 0.5):
            raise ConfigError(
                f"alt_yield out of range (0.0, 0.5) — got {config.screener.alt_yield}; likely unit confusion"
            )
        if not (0.0 < config.screener.fill_loss_coarse_bound < 0.2):
            raise ConfigError(
                f"fill_loss_coarse_bound out of range (0.0, 0.2) — got {config.screener.fill_loss_coarse_bound}; likely unit confusion"
            )

    def compute_r_apr_on_collateral(
        *,
        e_liq_per_epoch: float,     # PENDLE/epoch from reward_budget_history.e_liq_per_epoch
        side_split: float,          # long_budget_share or short_budget_share (0.0..1.0)
        q_me: float,                # operator-committed collateral-equivalent ($)
        sigma_others: float,        # aggregate competing size in range ($)
        t_frac: float,              # time-frac of epoch posted (v0.5: 1.0)
        pendle_price_usd: float,    # knob — operator supplies (PENDLE → USD)
        required_collateral: float, # max($100, Q_min × maintenance_margin); v0.5 uses 100
    ) -> float:
        """Returns R_apr_on_collateral (decimal fraction, e.g. 0.15 = 15% APR)."""
        if sigma_others + q_me * t_frac == 0:
            return 0.0
        r_share_per_epoch_pendle = e_liq_per_epoch * side_split * (q_me * t_frac) / (sigma_others + q_me * t_frac)
        r_share_per_epoch_usd = r_share_per_epoch_pendle * pendle_price_usd
        r_apr = r_share_per_epoch_usd * EPOCHS_PER_YEAR / required_collateral
        return r_apr

    def run_economics_gate(
        config_path: str,
        screener_metrics_path: str = "var/reports/screener_metrics.json",
    ) -> list[str]:
        """Applies DATA-09 kill-switch. Returns shortlist. Raises ProjectHalt on empty."""
        raw = load_raw_yaml(config_path)
        _assert_knobs_explicit(raw)                 # fail loud on missing
        config = load_config(config_path)
        _assert_knobs_in_range(config)              # fail loud on unit confusion

        metrics = json.loads(Path(screener_metrics_path).read_text())
        threshold = config.screener.alt_yield + config.screener.fill_loss_coarse_bound

        rows = []
        shortlist = []
        for mr in metrics["results"]:
            if not mr["filter_pass"]:
                continue   # already rejected by DATA-08; DATA-09 only considers passers
            # For each candidate, recompute R_apr from most-recent reward_budget_history row.
            # (Or use mr["r_apr_potential"] computed by plan 01-11.)
            r_apr = mr.get("r_apr_potential")
            if r_apr is None:
                log.warning("economics_gate.no_r_apr", market_id=mr["market_id"])
                continue
            delta = r_apr - threshold
            passes = delta >= 0
            rows.append({
                "market_id": mr["market_id"],
                "r_apr_on_collateral": r_apr,
                "alt_yield_used": config.screener.alt_yield,
                "fill_loss_used": config.screener.fill_loss_coarse_bound,
                "threshold": threshold,
                "delta": delta,
                "halt_reason": "" if passes else f"R_apr {r_apr:.4f} < threshold {threshold:.4f}",
            })
            if passes:
                shortlist.append(mr["market_id"])

        # Always write per-market decisions to a JSON report (non-halt audit trail)
        Path("var/reports").mkdir(parents=True, exist_ok=True)
        Path("var/reports/economics_gate.json").write_text(json.dumps({
            "threshold": threshold,
            "rows": rows,
            "shortlist": shortlist,
        }, indent=2))

        if not shortlist:
            # Empty shortlist → halt + kill report
            write_kill_report(rows, config=config, threshold=threshold)
            raise ProjectHalt(
    ```

    ```python
    """DATA-10 daily re-screen.

    Addresses PITFALLS §A5 (Competition crowding in during engineering window):
    DATA-10 runs under launchd CalendarInterval while Phase 2+ are building the models,
    so shortlist shifts are caught within 24h — operator knows if a market passed the
    economics gate yesterday but fails today, or if a new market has entered the candidate pool.

    Pipeline: ingest NDJSON deltas → run screener (DATA-08) → run economics gate (DATA-09) →
    diff today's shortlist vs yesterday's → write rescreen_diff.json.
    """
    import json, sys
    from datetime import datetime, timezone
    from pathlib import Path
    from structlog import get_logger
    from ..config.loader import load_config
    from ..manifest.writer import build_run_manifest, emit_manifest
    from ..data.ingester import market_trades, order_book, underlying_apr, settlement_md
    from .cli import run_screener    # programmatic screener from plan 01-11
    from .economics_gate import run_economics_gate, ProjectHalt, ConfigError

    log = get_logger(__name__)

    def run_daily_rescreen(config_path: str = "strategy/v0.5-default.yaml") -> int:
        config = load_config(config_path)
        manifest = build_run_manifest(config)
        emit_manifest(manifest)

        # 1. Ingest new NDJSON files
        log.info("rescreen.stage.ingest.start")
        for ingester in (market_trades, order_book, underlying_apr, settlement_md):
            ingester.ingest_recent_deltas("data/raw/", "data/parquet/", config)
        log.info("rescreen.stage.ingest.done")

        # 2. Run screener
        log.info("rescreen.stage.screener.start")
        run_screener(config)   # writes var/reports/screener_metrics.json
        log.info("rescreen.stage.screener.done")

        # 3. Load yesterday's shortlist (if exists) for diff
        yesterday_path = Path("var/reports/shortlist_previous.json")
        yesterday_shortlist = set()
        if yesterday_path.exists():
            yesterday_shortlist = set(json.loads(yesterday_path.read_text()).get("markets", []))

        # 4. Run economics gate
        try:
            today_shortlist = set(run_economics_gate(config_path))
        except ProjectHalt:
            today_shortlist = set()
            _write_rescreen_diff(yesterday_shortlist, today_shortlist)
            raise   # propagate exit code 1
        except ConfigError:
            raise   # propagate exit code 2

        # 5. Diff
        _write_rescreen_diff(yesterday_shortlist, today_shortlist)
        yesterday_path.write_text(json.dumps({"markets": sorted(today_shortlist)}, indent=2))
        return 0

    def _write_rescreen_diff(yesterday: set[str], today: set[str]) -> None:
        new_entrants = sorted(today - yesterday)
        dropped = sorted(yesterday - today)
        unchanged = sorted(today & yesterday)
        for m in dropped:
            log.warning("rescreen.market_dropped", market_id=m)
        for m in new_entrants:
            log.info("rescreen.market_entered", market_id=m)
        diff = {
            "timestamp_utc": datetime.now(timezone.utc).isoformat(),
            "yesterday_shortlist": sorted(yesterday),
            "today_shortlist": sorted(today),
            "new_entrants": new_entrants,
            "dropped": dropped,
            "unchanged": unchanged,
        }
        Path("var/reports").mkdir(parents=True, exist_ok=True)
        Path("var/reports/rescreen_diff.json").write_text(json.dumps(diff, indent=2))

    if __name__ == "__main__":
        try:
            sys.exit(run_daily_rescreen())
        except ProjectHalt:
            sys.exit(1)
        except ConfigError:
            sys.exit(2)
    ```
