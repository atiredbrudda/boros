    ```python
    """A8 mitigation: per-market SCALE calibration.

    SCALE connects the packed fIndexOracle's floatingIndex_raw to display APR:
        fIndex_display = 10000 + floatingIndex_raw / SCALE
    One value works for ETHUSDT (≈ 5.591e12), but other markets vary. We calibrate from
    two observed (cloThresh, fIndex) pairs ≥ 1 day apart.
    """
    from dataclasses import dataclass
    from structlog import get_logger
    log = get_logger(__name__)

    class InsufficientCalibrationData(RuntimeError): ...

    @dataclass(frozen=True)
    class CalibrationPoint:
        snapshot_ts: int              # unix ts — matches Wave-0 column name (01-02)
        clo_lower_thresh: float       # matches Wave-0 column name
        clo_upper_thresh: float       # matches Wave-0 column name
        f_index_raw: int              # decoded floatingIndex from bytes26 (matches 01-02 schema + 01-09 population)
        market_id: str

    @dataclass(frozen=True)
    class ScaleResult:
        scale: float
        scale_unstable: bool     # True if the two pairs yield divergent SCALE > 1%

    def calibrate_scale(p1: CalibrationPoint, p2: CalibrationPoint) -> ScaleResult:
        # From the fIndex equation: f_index_raw = SCALE * (fIndex_display - 10000)
        # The CLO threshold is the fIndex_display at which reward eligibility starts/ends;
        # assume clo_lower_thresh corresponds to a known-good fIndex anchor (operator supplies per market).
        # Procedure: for each point, SCALE_i = p.f_index_raw / (p.clo_lower_thresh - 10000)
        # Then average and flag if divergent.
        s1 = p1.f_index_raw / (p1.clo_lower_thresh - 10000.0)
        s2 = p2.f_index_raw / (p2.clo_lower_thresh - 10000.0)
        scale_avg = (s1 + s2) / 2.0
        divergence = abs(s1 - s2) / scale_avg if scale_avg != 0 else float("inf")
        unstable = divergence > 0.01
        if unstable:
            log.warning("screener.scale_unstable", market_id=p1.market_id,
                        scale_1=s1, scale_2=s2, divergence_pct=divergence * 100)
        return ScaleResult(scale=scale_avg, scale_unstable=unstable)

    def get_calibration_pairs(conn, market_id: str) -> tuple[CalibrationPoint, CalibrationPoint]:
        """Query clo_history for 2 rows ≥ 1 day apart with distinct (clo_lower_thresh, f_index_raw).

        Columns queried match the Wave-0 canonical schema (01-02) with the f_index_raw addition:
        `snapshot_ts, clo_lower_thresh, clo_upper_thresh, f_index_raw`. Populated by 01-09
        MarketsPoller every poll. Per checker issue #05.
        """
        rows = conn.execute("""
            SELECT snapshot_ts, clo_lower_thresh, clo_upper_thresh, f_index_raw
            FROM clo_history
            WHERE market_id = ? AND f_index_raw IS NOT NULL
            ORDER BY snapshot_ts ASC
        """, (market_id,)).fetchall()
        if len(rows) < 2:
            raise InsufficientCalibrationData(
                f"{market_id}: need >= 2 clo_history rows, have {len(rows)}. Wait ≥ 1 day."
            )
        # Pick first + last for max time spread
        first, last = rows[0], rows[-1]
        # Assert ≥ 86400s apart
        ...
        return (CalibrationPoint(...), CalibrationPoint(...))
    ```

    ```python
    from typing import Optional
    from .metrics import (
        compute_inter_arrival_cv, compute_realized_vol, compute_reversal_count,
        compute_depth_at_clo_edge, compute_outside_range_wall_ratio,

    @dataclass
    class MarketScreeningResult:
        market_id: str
        cv: Optional[float]
        realized_vol: Optional[float]
        reversal_count: Optional[int]
        depth_at_clo_edge: Optional[float]
        outside_range_wall_ratio: Optional[float]
        outside_range_wall_flag: bool
        clo_lower: float
        clo_upper: float
        time_to_maturity_days: float
        scale: Optional[float]
        scale_unstable: bool
        r_apr_potential: Optional[float]
        filter_pass: bool
        filter_failure_reasons: list[str] = field(default_factory=list)

    def apply_screener_filters(markets: list[str], duck_conn, config) -> list[MarketScreeningResult]:
        """DATA-08: apply the 6 filters per market. Ranks passing markets by r_apr_potential
        descending; returns shortlist capped at 4 (roadmap contract: 2-4 markets).

            trades = duck_conn.execute(
                f"SELECT * FROM market_trades WHERE market_id = ? AND datetime >= now() - INTERVAL {config.screener.trades_window_days} DAYS",
                (mid,)
            ).pl()
            cv = compute_inter_arrival_cv(trades, market_id=mid) if trades.height > 0 else None
            if cv is None: fails.append("cv_unavailable")

            rc = compute_reversal_count(trades, window_days=config.screener.reversal_count_window_days) if trades.height > 0 else None
            if rc is None or rc < 1: fails.append("insufficient_reversals")

            # Canonical Wave-0 columns: snapshot_ts, market_id, clo_lower_thresh, clo_upper_thresh, f_index_raw
            # (per checker issue #05). maturity_ts lives on MarketCalibration (config.markets), NOT clo_history.
            clo = duck_conn.execute(
                "SELECT clo_lower_thresh, clo_upper_thresh FROM clo_history WHERE market_id = ? ORDER BY snapshot_ts DESC LIMIT 1",
                (mid,)
            ).fetchone()
            if clo is None: fails.append("no_clo_history")
            clo_lower, clo_upper = (clo if clo else (0, 0))
            # maturity_ts comes from MarketCalibration
            maturity_ts = next((m.maturity_ts for m in config.markets if m.market_id == mid), 0)
            book = duck_conn.execute(
                "SELECT * FROM order_book_combined WHERE market_id = ? ORDER BY snapshot_ts DESC LIMIT 1000",
                (mid,)
            ).pl()
            depth = compute_depth_at_clo_edge(book, clo_lower, clo_upper) if book.height > 0 else None
            if depth is None or depth <= 0: fails.append("zero_depth_at_clo")
            wall_ratio = compute_outside_range_wall_ratio(book, clo_lower, clo_upper) if book.height > 0 else None
            # Filter 5: CLO bounds non-zero
            if clo_lower == 0 or clo_upper == 0: fails.append("clo_bounds_zero")
            # Filter 6: time to maturity
            ttm = compute_days_to_maturity(maturity_ts)
            if ttm < config.screener.min_time_to_maturity_days: fails.append(f"ttm_lt_{config.screener.min_time_to_maturity_days}d")
            # SCALE calibration (non-blocking — informational)

            try:
                p1, p2 = get_calibration_pairs(duck_conn.execute("SELECT DISTINCT * FROM clo_history"), mid)

                scale, scale_unstable = sr.scale, sr.scale_unstable

            # long_budget, short_budget, data_source. Populated by 01-10 (data_source IN ('polled','synthetic_current')).

                "SELECT e_liq_per_epoch FROM reward_budget_history WHERE market_id = ? ORDER BY snapshot_ts DESC LIMIT 1",
                (mid,)
            ).fetchone()
            r_apr_potential = (latest_budget[0] * 52 / 100.0) if latest_budget else None

            results.append(MarketScreeningResult(
                market_id=mid, cv=cv, realized_vol=rv, reversal_count=rc,
                depth_at_clo_edge=depth, outside_range_wall_ratio=wall_ratio,
                outside_range_wall_flag=(wall_ratio is not None and wall_ratio > 2.0),
                clo_lower=clo_lower, clo_upper=clo_upper, time_to_maturity_days=ttm,
                scale=scale, scale_unstable=scale_unstable, r_apr_potential=r_apr_potential,
                filter_pass=(len(fails) == 0), filter_failure_reasons=fails,
            ))

        passing.sort(key=lambda r: r.r_apr_potential or 0, reverse=True)
        shortlist = passing[:4]
        # Return ALL results (for report) with filter_pass set; caller filters to shortlist
        return results
    ```
