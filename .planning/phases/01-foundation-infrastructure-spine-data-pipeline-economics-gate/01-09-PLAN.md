```python
"""Shared httpx AsyncClient for Boros REST. Used by /markets poller (DATA-06) and
E_liq poller (DATA-11, Wave 2).
"""
from __future__ import annotations
import httpx
import structlog
from boros_brain.config.schema import BotConfig

log = structlog.get_logger(__name__)


class BorosApiError(RuntimeError):
    """Open-API-shape error: {errorCode, message, data}."""
    def __init__(self, *, status: int, errorCode: str | None, message: str):
        super().__init__(f"BorosApiError[{status}]: {errorCode}: {message}")
        self.status = status
        self.errorCode = errorCode
        self.message = message


def make_boros_rest_client(config: BotConfig, timeout_s: float = 10.0) -> httpx.AsyncClient:
    """Returns an httpx.AsyncClient pre-configured with base_url + timeout."""
    return httpx.AsyncClient(
        base_url=config.data.boros_rest_base,
        timeout=httpx.Timeout(timeout_s, connect=3.0),
        follow_redirects=False,
    )
```

```python
"""Single-writer for the clo_history table (INFRA-03 discipline). Only
markets_poller imports this module; other readers go through DuckDB.

Function name `insert_clo_history` matches the 01-02 Wave-0 writer convention
(insert_<table_name>). Column names are `snapshot_ts, clo_lower_thresh, clo_upper_thresh,
f_index_raw` per the Wave-0 canonical schema — NOT camelCase API wire names.
"""
from __future__ import annotations
from pathlib import Path
import sqlite3
import structlog
from boros_brain.state.db import open_connection

log = structlog.get_logger(__name__)


def insert_clo_history(
    sqlite_path: Path | str,
    *,
    market_id: str,
    snapshot_ts: int,
    clo_lower_thresh: float,
    clo_upper_thresh: float,
    f_index_raw: int | None,
) -> None:
    """Append one row to clo_history. Caller has already determined this is a CHANGE
    relative to the last observed row for this market_id. f_index_raw is the decoded
    floatingIndex_raw from the fIndexOracle bytes26 payload (nullable if not yet decoded
    for a pre-fIndex-support historical row; MarketsPoller always supplies a value).
    """
    with open_connection(sqlite_path) as conn:
        conn.execute(
            "INSERT INTO clo_history "
            "(snapshot_ts, market_id, clo_lower_thresh, clo_upper_thresh, f_index_raw) "
            "VALUES (?, ?, ?, ?, ?)",
            (snapshot_ts, market_id, clo_lower_thresh, clo_upper_thresh, f_index_raw),
        )
    log.info(
        "clo_history.appended",
        market_id=market_id,
        snapshot_ts=snapshot_ts,
        clo_lower_thresh=clo_lower_thresh,
        clo_upper_thresh=clo_upper_thresh,
        f_index_raw=f_index_raw,
    )
```

```python
        self._last: dict[str, tuple[float, float, int | None]] = {}
        self._findex_readers: dict[str, FIndexReader] = dict(findex_readers or {})
        self._backoff_base_s = backoff_base_s

    async def poll_once(self) -> list[Market]:
        """One HTTP call + schema validation. Retries on 5xx with exponential backoff.
        Raises on 4xx or unexpected shape."""
        attempts = 3
        for attempt in range(attempts):
            try:
                resp = await self.http.get("/markets")
                if resp.status_code == 200:
                    data = MarketsResponse.model_validate_json(resp.content)
                    return data.markets
                if 500 <= resp.status_code < 600:
                    raise httpx.HTTPStatusError(f"5xx", request=resp.request, response=resp)
                resp.raise_for_status()
            except (httpx.HTTPError, httpx.HTTPStatusError) as e:
                if attempt == attempts - 1:
                    raise
                delay = self._backoff_base_s * (2 ** attempt) + random.uniform(0, self._backoff_base_s * 0.5)
                log.warning("markets_poller.transient_error", attempt=attempt, delay_s=round(delay, 3), err=str(e))
                await asyncio.sleep(delay)
        return []  # unreachable

    async def tick(self) -> int:
        """Run one poll + write changes. Returns number of rows written.

        For each market we:
          1) parse clo_lower_thresh, clo_upper_thresh from /markets
          2) decode floatingIndex_raw from the per-market fIndexOracle (DATA-07)
          3) compare (clo_lower, clo_upper, f_index_raw) tuple against previous snapshot
          4) insert_clo_history ONLY on change (keeps clo_history compact)
        """
        markets = await self.poll_once()
        snapshot_ts = await now_cached()
        written = 0
        for m in markets:
            # Decode fIndex for this market. If the market has no fIndexOracle address
            # (pre-launch or non-incentivized row), record f_index_raw=None.
            f_index_raw: int | None = None
            if m.fIndexOracle:
                reader = self._findex_readers.get(m.marketId)
                if reader is None:
                    # Lazy-create per-market reader (shared RPC client would be an
                    # optimisation; for Phase 1 we accept one connection per market).
                    w3 = await make_arbitrum_rpc(self.config)
                    reader = FIndexReader(
                        w3=w3,
                        oracle_address=m.fIndexOracle,
                        ttl_s=self.config.data.chain_time_cache_ttl_s,
                    )
                    self._findex_readers[m.marketId] = reader
                try:
                    payload = await reader.get_latest_fIndex()
                    f_index_raw = int(payload["floatingIndex"])
                except Exception as e:
                    # Decode/RPC failure is non-fatal — we still record CLO thresholds.
                    log.warning(
                        "markets_poller.findex_decode_failed",
                        market_id=m.marketId,
                        err=str(e),
                    )

            prev = self._last.get(m.marketId)
            cur = (m.cloLowerThresh, m.cloUpperThresh, f_index_raw)
            if prev != cur:
                insert_clo_history(
                    sqlite_path=self.config.infra.sqlite_path,
                    market_id=m.marketId,
                    snapshot_ts=snapshot_ts,
                    clo_lower_thresh=cur[0],
                    clo_upper_thresh=cur[1],
                    f_index_raw=cur[2],
                )
                self._last[m.marketId] = cur
                written += 1
        log.info("markets_poller.tick_done", total=len(markets), written=written)
        return written


async def run_markets_poller(config: BotConfig, run_id: str) -> None:
    """Long-running task — call from brain main.py."""
    async with make_boros_rest_client(config) as http:
        poller = MarketsPoller(config, http, run_id)
        while True:
            try:
                await poller.tick()
            except Exception as e:
                log.exception("markets_poller.tick_failed", err=str(e))
            await asyncio.sleep(config.data.markets_poll_interval_s)
```

```python
from dataclasses import dataclass, field
from typing import Optional
from web3 import AsyncWeb3
import structlog

log = structlog.get_logger(__name__)


FINDEX_ABI = [
    {"name": "getLatestFIndex",      "inputs": [], "outputs": [{"type": "bytes32"}], "stateMutability": "view", "type": "function"},
    {"name": "latestAnnualizedRate", "inputs": [], "outputs": [{"type": "int256"}],  "stateMutability": "view", "type": "function"},

class FIndexDecodeError(RuntimeError):
    """Raised when decoded fIndex fields fail sanity bounds."""


def decode_findex_bytes(raw: bytes) -> dict:
    """Pure-function decoder for the bytes32 payload returned by getLatestFIndex().
    Per boros_bot_guide.md L202-213 (verified 2026-04-23 via grep in RESEARCH.md).

    Layout (bottom 26 bytes = 208 bits of the bytes32):
        bits [176..208): fTime (uint32, unix timestamp)
        bits [ 64..176): floatingIndex (int112, 1e18 precision, SIGN-EXTENDED)
        bits [  0..64 ): feeIndex (uint64)
    """
    if len(raw) != 32:
        raise ValueError(f"fIndex payload must be 32 bytes, got {len(raw)}")
    val208 = int.from_bytes(raw[-26:], "big")
    fTime = (val208 >> 176) & ((1 << 32) - 1)
    floatingIndex = (val208 >> 64) & ((1 << 112) - 1)
    if floatingIndex >= (1 << 111):
        floatingIndex -= (1 << 112)              # int112 sign extension
    feeIndex = val208 & ((1 << 64) - 1)

    # Sanity bounds (post-decode) — see T-01-09-02 threat.
    if abs(floatingIndex) >= (1 << 110):
        raise FIndexDecodeError(
            f"floatingIndex={floatingIndex} exceeds 2^110; likely off-by-one or wrong-endian decode"
        )
    return {"fTime": fTime, "floatingIndex": floatingIndex, "feeIndex": feeIndex}


def fIndex_display(floatingIndex: int, scale: float) -> float:
    """Per research §Pattern 8 line 1021: fIndex_display = 10000 + floatingIndex / SCALE.
    SCALE is per-market (≈ 5.591e12 for Binance ETHUSDT 29-May-2026 per research).
    """
    return 10000.0 + floatingIndex / scale


@dataclass
class FIndexReader:
    w3: AsyncWeb3
    oracle_address: str
    ttl_s: float = 1.0
    _cache: Optional[dict] = None
    _ts: float = 0.0

    def _contract(self):
        return self.w3.eth.contract(address=self.oracle_address, abi=FINDEX_ABI)

    async def get_latest_fIndex(self) -> dict:
        now = time.monotonic()
        if self._cache is not None and (now - self._ts) < self.ttl_s:
            return self._cache
        raw = await self._contract().functions.getLatestFIndex().call()  # bytes32
        result = decode_findex_bytes(raw)
        self._cache = result
        self._ts = now
        return result

    async def latest_annualized_rate(self) -> int:
        """Returns 1e18-precision signed int. Caller divides by 1e18 for APR float."""
        return await self._contract().functions.latestAnnualizedRate().call()
```
