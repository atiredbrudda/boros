```python
"""Shared ingester primitives."""
from __future__ import annotations
from dataclasses import dataclass, field
from pathlib import Path
from typing import Iterable
import polars as pl
import structlog

log = structlog.get_logger(__name__)


class IngesterSchemaViolation(RuntimeError):
    """Raised when NDJSON row fails a hard-fail schema assertion (e.g., missing eventIndex)."""


@dataclass(frozen=True)
class IngestResult:
    rows_in: int
    rows_written: int
    path_out: str
    eventIndex_missing: int = 0
    assertion_failures: int = 0


def write_partitioned_parquet(
    df: pl.DataFrame,
    out_root: Path,
    partition_cols: list[str],
    subdir: str,
) -> Path:
    """Write DataFrame as Hive-partitioned parquet under out_root/subdir/.
    Uses pyarrow backend for Hive partition_cols support per polars docs.
    Returns the absolute path written to.
    """
    missing = [c for c in partition_cols if c not in df.columns]
    if missing:
        raise ValueError(f"partition_cols {missing} not present in df columns {df.columns}")
    full = Path(out_root) / subdir
    full.mkdir(parents=True, exist_ok=True)
    df.write_parquet(
        str(full),
        use_pyarrow=True,
        pyarrow_options={"partition_cols": partition_cols},
    )
    log.info("ingester.parquet_written", path=str(full), rows=df.height, partitions=partition_cols)
    return full
```

    4. Write `python/src/boros_brain/data/ingester/market_trades.py`:
```python
"""DATA-01 — market-trades NDJSON ingester.

Per CLAUDE.md: preserve eventIndex (intra-block ordering primitive — NOT blockTimestamp).
Per PITFALL §B2 (research §Pitfall 6): eventIndex null/missing → raise IngesterSchemaViolation.
Per DATA-01 spec: wei→float conversion is EXPLICIT (no silent truncation).
"""
from __future__ import annotations
from pathlib import Path
import polars as pl
import structlog
from .base import IngestResult, IngesterSchemaViolation, write_partitioned_parquet
from .schemas import MARKET_TRADES_SCHEMA

log = structlog.get_logger(__name__)


def _parse_size_column(df: pl.DataFrame) -> pl.DataFrame:
    """size may arrive as wei-string ('5000000000000000') OR as a JSON number coerced by scan_ndjson.
    Emit two columns: size_wei:UInt128 (nullable) + size_float:Float64.
    If size parses as int-like → wei path. If it's decimal-like → float path (wei=None).
    """
    # Attempt wei-string cast
    try:
        df2 = df.with_columns(
            pl.col("size").cast(pl.UInt128, strict=True).alias("size_wei"),
        ).with_columns(
            (pl.col("size_wei").cast(pl.Float64) / 1e18).alias("size_float"),
        )
        log.info("ingester.market_trades.size_path", path="wei_string")
        return df2.drop("size")
    except (pl.exceptions.InvalidOperationError, pl.exceptions.ComputeError):
        # Fallback: size is already a float (JSON number in NDJSON)
        df2 = df.with_columns(
            pl.col("size").cast(pl.Float64, strict=False).alias("size_float"),
            pl.lit(None).cast(pl.UInt128).alias("size_wei"),
        )
        log.info("ingester.market_trades.size_path", path="float_direct")
        return df2.drop("size")


def ingest_market_trades(
    ndjson_path: Path, market_id: str, out_root: Path
) -> IngestResult:
    """Ingest one market-trades NDJSON file into Hive-partitioned parquet.
    Raises IngesterSchemaViolation if ANY row has null/missing eventIndex (PITFALL §B2).
    """
    df = pl.read_ndjson(str(ndjson_path), schema=MARKET_TRADES_SCHEMA)
    rows_in = df.height

    # PITFALL §B2 hard-fail: eventIndex must be present + non-null on every row
    null_count = int(df["eventIndex"].null_count())
    if null_count > 0:
        raise IngesterSchemaViolation(
            f"{ndjson_path.name}: {null_count}/{rows_in} rows have null/missing eventIndex "
            f"(market_id={market_id}). DATA-01 requires eventIndex on every row."
        )

    # Side constraint check (soft: log counts, don't raise)
    invalid_side = df.filter(~pl.col("side").is_in(["long", "short"])).height
    if invalid_side:
        log.warning("ingester.market_trades.invalid_side_count", count=invalid_side)

    df = _parse_size_column(df)
    # Add partition columns
    df = df.with_columns(
        pl.lit(market_id).alias("market_id"),
        pl.from_epoch(pl.col("blockTimestamp"), time_unit="s").dt.strftime("%Y-%m").alias("year_month"),
    )

    path_out = write_partitioned_parquet(
        df,
        out_root=out_root,
        partition_cols=["market_id", "year_month"],
        subdir="market_trades",
    )
    return IngestResult(
        rows_in=rows_in,
        rows_written=df.height,
        path_out=str(path_out),
        eventIndex_missing=0,
    )
```

```python
"""DATA-02 — order-book ingester. Two shapes:
- raw: per-tick snapshot of the native Boros order book.
- combined_{tickSize}: book merged with AMM liquidity for a given tick size.

The two shapes share (snapshot_ts, tick) as the join key so DATA-05 AMM conservation
(Wave 2) can compute combined - raw >= 0 per tick.
"""
from __future__ import annotations
from pathlib import Path
import polars as pl
import structlog
from .base import IngestResult, write_partitioned_parquet
from .schemas import ORDER_BOOK_RAW_SCHEMA, ORDER_BOOK_COMBINED_SCHEMA

log = structlog.get_logger(__name__)


def _parse_wei_size(df: pl.DataFrame, col: str, out_wei: str, out_float: str) -> pl.DataFrame:
    """Convert wei-string column to UInt128 + Float64 derived."""
    return df.with_columns(
        pl.col(col).cast(pl.UInt128, strict=False).alias(out_wei),
    ).with_columns(
        (pl.col(out_wei).cast(pl.Float64) / 1e18).alias(out_float),
    ).drop(col)


def ingest_order_book_raw(
    ndjson_path: Path, market_id: str, out_root: Path
) -> IngestResult:
    """Ingest a single hourly raw snapshot NDJSON."""
    df = pl.read_ndjson(str(ndjson_path), schema=ORDER_BOOK_RAW_SCHEMA)
    rows_in = df.height

    df = _parse_wei_size(df, "sz", "sz_wei", "sz_float")
    df = df.with_columns(pl.lit(market_id).alias("market_id"))

    path_out = write_partitioned_parquet(
        df, out_root=out_root,
        partition_cols=["market_id", "snapshot_ts"],
        subdir="order_book_raw",
    )
    return IngestResult(
        rows_in=rows_in, rows_written=df.height, path_out=str(path_out),
    )


def ingest_order_book_combined(
    ndjson_path: Path, market_id: str, tick_size: str, out_root: Path
) -> IngestResult:
    """Ingest a single combined-book snapshot NDJSON for a given tick_size (e.g., '0.01')."""
    df = pl.read_ndjson(str(ndjson_path), schema=ORDER_BOOK_COMBINED_SCHEMA)
    rows_in = df.height

    df = _parse_wei_size(df, "sz_combined", "sz_combined_wei", "sz_combined_float")
    df = df.with_columns(
        pl.lit(market_id).alias("market_id"),
        pl.lit(tick_size).alias("tick_size"),
    )

    path_out = write_partitioned_parquet(
        df, out_root=out_root,
        partition_cols=["market_id", "tick_size", "snapshot_ts"],
        subdir="order_book_combined",
    )
    return IngestResult(
        rows_in=rows_in, rows_written=df.height, path_out=str(path_out),
    )
```

```python
"""DATA-04 — settlement + market-data NDJSON ingester. Two shapes written to separate parquet subdirs."""
from __future__ import annotations
from pathlib import Path
from typing import Optional
import polars as pl
from .base import IngestResult, write_partitioned_parquet
from .schemas import SETTLEMENT_SCHEMA, MARKET_DATA_SCHEMA


def _ingest_settlement_one(path: Path, market_id: str, out_root: Path) -> IngestResult:
    df = pl.read_ndjson(str(path), schema=SETTLEMENT_SCHEMA)
    rows_in = df.height
    df = df.with_columns(
        pl.from_epoch(pl.col("blockTimestamp"), time_unit="s").dt.strftime("%Y-%m").alias("year_month"),
    )
    path_out = write_partitioned_parquet(
        df, out_root=out_root,
        partition_cols=["market_id", "year_month"],
        subdir="settlement",
    )
    return IngestResult(rows_in=rows_in, rows_written=df.height, path_out=str(path_out))


def _ingest_market_data_one(path: Path, market_id: str, out_root: Path) -> IngestResult:
    df = pl.read_ndjson(str(path), schema=MARKET_DATA_SCHEMA)
    rows_in = df.height
    df = df.with_columns(
        pl.from_epoch(pl.col("blockTimestamp"), time_unit="s").dt.strftime("%Y-%m").alias("year_month"),
    )
    path_out = write_partitioned_parquet(
        df, out_root=out_root,
        partition_cols=["market_id", "year_month"],
        subdir="market_data",
    )
    return IngestResult(rows_in=rows_in, rows_written=df.height, path_out=str(path_out))


def ingest_settlement_md(
    settlement_path: Optional[Path],
    market_data_path: Optional[Path],
    market_id: str,
    out_root: Path,
) -> dict[str, IngestResult]:
    """Ingest settlement + market-data NDJSON files into separate parquet subdirs.
    Both args may be None (skip that shape). Returns {'settlement': IngestResult, 'market_data': IngestResult}.
    """
    result = {}
    if settlement_path is not None:
        result["settlement"] = _ingest_settlement_one(settlement_path, market_id, out_root)
    if market_data_path is not None:
        result["market_data"] = _ingest_market_data_one(market_data_path, market_id, out_root)
    return result
```

```python
"""CLI: python -m boros_brain.data.ingester.cli --kind <k> --src <path> [--market-id X] [--exchange Y] [--tick-size Z] --out <parquet_root>"""
from __future__ import annotations
import argparse, sys
from pathlib import Path
import structlog
from boros_brain.config.loader import load_bot_config
from boros_brain.logging.structlog_setup import configure_structlog
from boros_brain.manifest.writer import write_run_manifest
from .market_trades import ingest_market_trades
from .order_book import ingest_order_book_raw, ingest_order_book_combined
from .underlying_apr import ingest_underlying_apr
from .settlement_md import ingest_settlement_md


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(description="Boros NDJSON ingester — DATA-01..04")
    p.add_argument("--kind", required=True,
                   choices=["market_trades", "order_book_raw", "order_book_combined",
                            "underlying_apr", "settlement_md"])
    p.add_argument("--src", required=True, type=Path, help="NDJSON path")
    p.add_argument("--market-id")
    p.add_argument("--exchange")
    p.add_argument("--tick-size")
    p.add_argument("--market-data-src", type=Path, help="For settlement_md: market-data NDJSON")
    p.add_argument("--out", required=True, type=Path, help="parquet root")
    p.add_argument("--config", type=Path, default=Path("strategy/v0.5-default.yaml"))
    args = p.parse_args(argv)

    config = load_bot_config(args.config, mode_override="ingest")
    run_id = write_run_manifest(config)
    configure_structlog(run_id=run_id, tier="brain")
    log = structlog.get_logger("ingester.cli")

    log.info("ingester.cli.start", kind=args.kind, src=str(args.src), out=str(args.out))

    if args.kind == "market_trades":
        assert args.market_id, "--market-id required"
        r = ingest_market_trades(args.src, args.market_id, args.out)
    elif args.kind == "order_book_raw":
        assert args.market_id
        r = ingest_order_book_raw(args.src, args.market_id, args.out)
    elif args.kind == "order_book_combined":
        assert args.market_id and args.tick_size
        r = ingest_order_book_combined(args.src, args.market_id, args.tick_size, args.out)
    elif args.kind == "underlying_apr":
        assert args.exchange
        r = ingest_underlying_apr(args.src, args.exchange, args.out)
    elif args.kind == "settlement_md":
        assert args.market_id
        r = ingest_settlement_md(args.src, args.market_data_src, args.market_id, args.out)
    else:
        raise ValueError(f"unknown kind: {args.kind}")

    print(f"ingested kind={args.kind} result={r}")
    log.info("ingester.cli.done", result=str(r))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
