    import polars as pl
    from .exceptions import AMMConservationError

    def assert_amm_conservation(raw: pl.DataFrame, combined: pl.DataFrame) -> None:
        """DATA-05: for each tick δ in an hourly snapshot, combined_size(δ) >= raw_book_size(δ).
        Violation => raise AMMConservationError. Formula: combined_size(δ) − raw_book_size(δ) ≥ 0 per tick."""
        joined = raw.join(

            on=["snapshot_ts", "tick"],
            how="outer",
            coalesce=True,
            suffix="_combined",
        )
        neg = joined.filter(pl.col("size_combined") - pl.col("size") < 0)
        if neg.height > 0:
            raise AMMConservationError(
                f"AMM conservation violated on {neg.height} tick-rows; "
                f"first 5: {neg.head(5).to_dicts()}"
            )

    ```python
    import asyncio
    from structlog import get_logger
    from ..rest_client import BorosRestClient
    from ..chain_time import get_chain_time_cached
    from ...state.writers.reward_budget_history import insert_reward_budget_row

    log = get_logger(__name__)

    class ELiqPoller:
        """DATA-11: polls GET /markets every e_liq_poll_interval_s (default 5s, range 1-5s)
        and writes one reward_budget_history row per market per cycle. NO CACHE — per
        ARCHITECTURE.md §9 Anti-pattern 4 / PITFALLS §A2, E_liq_per_epoch rebalances
        dynamically; a stale cache produces wrong R_apr_on_collateral in DATA-09."""

        def __init__(self, config, conn, rest, run_id, event_pub):
            self.interval = config.data.e_liq_poll_interval_s   # 1-5s per DataConfig
            self.markets = [m.market_id for m in config.markets]  # v0.5: single market
            self.conn = conn
            self.rest = rest
            self.run_id = run_id
            self.event_pub = event_pub
            self._fail_count = 0

        async def run_one_cycle(self) -> None:
            try:
                markets = await self._get_with_backoff()
            except Exception as e:
                self._fail_count += 1
                log.error("e_liq_poller.fetch_failed", error=str(e), fail_count=self._fail_count)
                if self._fail_count >= 5:
                    await self.event_pub.publish("poller.e_liq.degraded",
                                                 {"fail_count": self._fail_count})
                return
            self._fail_count = 0
            chain_ts = await get_chain_time_cached()
            for m in markets:
                if m["marketId"] not in self.markets:
                    continue
                # Field names per assumption A5 — reconcile with live response in Wave-0 spike
                insert_reward_budget_row(
                    self.conn,
                    market_id=m["marketId"],
                    e_liq_per_epoch=float(m["liquidityRewardPerEpoch"]),
                    long_budget=float(m["longBudget"]),
                    short_budget=float(m["shortBudget"]),
                    chain_ts=chain_ts,
                    run_id=self.run_id,
                    data_source="polled",
                )

        async def _get_with_backoff(self):
            """httpx with exponential backoff: 0.5, 1, 2, 4, 8s cap — 5 retries."""
            delays = [0.5, 1.0, 2.0, 4.0, 8.0]
            last_exc = None
            for d in delays:
                try:
                    return await self.rest.get_markets()
                except Exception as e:
                    last_exc = e
                    await asyncio.sleep(d)
            raise last_exc

        async def loop(self) -> None:
            while True:
                await self.run_one_cycle()
                await asyncio.sleep(self.interval)
    ```
