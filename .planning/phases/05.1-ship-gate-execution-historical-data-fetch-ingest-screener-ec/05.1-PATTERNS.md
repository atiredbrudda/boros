#### Pattern 5 — Atomic write + sha256 manifest entry (NEW — no exact analog; closest is `manifest/writer.py:101-105` for sidecar atomic-write style)

```python
import hashlib
import os
from datetime import UTC, datetime

def _sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()

def _atomic_write(local_path: Path, body: bytes) -> None:
    local_path.parent.mkdir(parents=True, exist_ok=True)
    tmp = local_path.with_suffix(local_path.suffix + ".tmp")
    tmp.write_bytes(body)
    os.replace(tmp, local_path)
```

**Keep:** `os.replace` (POSIX atomic rename). 1 MiB streaming hash chunk (avoids loading multi-GB files into RAM).
**Change:** ZIP-bomb cap — wrap extraction in a per-file `if extracted_size > MAX_DECOMPRESSED_BYTES: raise` guard (RESEARCH.md §Security row 1). Use `MAX_DECOMPRESSED_BYTES = 10 * 1024**3` (10 GB).

#### Pattern 6 — Idempotency manifest JSON shape (NEW — defined in RESEARCH.md §Idempotency contract)

```python
import json

def _load_manifest(path: Path) -> dict:
    if not path.exists():
        return {"fetched": []}
    return json.loads(path.read_text())

def _is_already_fetched(manifest: dict, url: str, local_path: Path) -> bool:
    """Skip rule: file exists locally AND manifest has matching (url, sha256, length) AND on-disk sha matches."""
    if not local_path.exists():
        return False
    for entry in manifest["fetched"]:
        if entry["url"] != url:
            continue
        if entry["local_path"] != str(local_path):
            continue
        # Re-hash the on-disk file — if tampered or partially overwritten, refetch.
        if _sha256_file(local_path) != entry["sha256"]:
            return False
        if local_path.stat().st_size != entry["content_length_bytes"]:
            return False
        return True
    return False

def _record_fetched(manifest: dict, *, url: str, local_path: Path,
                    content_length: int, sha256: str, run_id: str) -> None:
    manifest["fetched"].append({
        "url": url,
        "local_path": str(local_path),
        "content_length_bytes": content_length,
        "sha256": sha256,
        "fetched_at_utc": datetime.now(UTC).isoformat(),
        "run_id": run_id,
    })
```

**Keep:** atomic manifest write at end of run (`tmp = path.with_suffix(".tmp"); tmp.write_text(json.dumps(...)); os.replace(tmp, path)`).
**Change:** none — this is fully new; the spec is in RESEARCH.md §Idempotency contract verbatim.

    tmp_path: Path,
    default_config: BotConfig,
    monkeypatch: pytest.MonkeyPatch,

    """Re-running the fetcher with a populated manifest + matching local files is a no-op."""

    manifest = {"fetched": []}
    local = tmp_path / "ndjson" / "market-trades" / "TEST" / "2026-04.ndjson"

#### Pattern 3 — ZIP-bomb cap test (NEW)

```python
def test_fetcher_rejects_zip_bomb(tmp_path: Path) -> None:
    """ZIP whose decompressed size exceeds MAX_DECOMPRESSED_BYTES halts with a clear error."""
    import zipfile
    from scripts.fetch_historical_data import _safe_extract_zip, MAX_DECOMPRESSED_BYTES

    bomb = tmp_path / "bomb.zip"
    # Construct a ZIP whose advertised uncompressed size > cap.
    # Simpler: monkeypatch a tiny cap and use a normal entry.
    import scripts.fetch_historical_data as fetcher_mod
    original_cap = fetcher_mod.MAX_DECOMPRESSED_BYTES
    try:
        fetcher_mod.MAX_DECOMPRESSED_BYTES = 64  # bytes
        with zipfile.ZipFile(bomb, "w") as zf:
            zf.writestr("payload.ndjson", b"x" * 1024)  # 1 KiB > 64 B cap

        with pytest.raises(ValueError, match="zip bomb|exceeds.*cap|max.*decompressed"):
            _safe_extract_zip(bomb, out_root=tmp_path / "ndjson")
    finally:
        fetcher_mod.MAX_DECOMPRESSED_BYTES = original_cap
```

**Keep:** `pytest.raises(..., match=...)` style.
**Change:** monkeypatching a module constant is a small ergonomic concession to keep the test fast (no need to actually fabricate a 10 GB ZIP). Acceptable per project test conventions (existing `test_chain_time_drift.py` and others monkeypatch module attrs similarly).
