    """

def enumerate_perturbable_leaves(
    config: BotConfig,
    *,
    exclude_prefixes: tuple[str, ...] = _PERTURBATION_EXCLUDE_PREFIXES,
) -> list[KnobLeaf]:
    """Walk config.model_dump(mode='json') for all numeric leaves.

    Numeric leaves = float | int (NOT bool, NOT str, NOT None). Per BACK-06,
    booleans (e.g., voluntary_cancel_forfeits_accumulated) are handled by the
    forfeit_branch sweep, not here. Per RESEARCH §Open Question #3, leaves whose
    dotted path starts with any `exclude_prefixes` entry are skipped.
    """
    out: list[KnobLeaf] = []

    def _walk(d, path: list[str]) -> None:
        # Mirrors tests/test_manifest_contains_all_knobs.py::_collect_leaf_paths:53-76
        if isinstance(d, dict):
            for k, v in d.items():
                _walk(v, path + [k])
        elif isinstance(d, list):
            for i, v in enumerate(d):
                _walk(v, path + [str(i)])
        elif isinstance(d, (int, float)) and not isinstance(d, bool):
            dotted = ".".join(path)
            if any(dotted.startswith(p) for p in exclude_prefixes):
                return
            out.append(KnobLeaf(path=dotted, value=float(d)))

    _walk(config.model_dump(mode="json"), [])
    return out


def detect_sign_flip(
    unperturbed_pnl: float, low_pnl: float, high_pnl: float
) -> bool:
    """RESEARCH.md L1036-1045 verbatim. Tri-valued sign extraction so 0.0 → 0
    and never spuriously flips."""
    sign_unp = (unperturbed_pnl > 0) - (unperturbed_pnl < 0)
    sign_low = (low_pnl > 0) - (low_pnl < 0)
    sign_high = (high_pnl > 0) - (high_pnl < 0)
    return sign_unp != sign_low or sign_unp != sign_high


def _set_leaf_by_path(d: dict, dotted: str, value: float) -> dict:
    """Walk dotted path through nested dicts/lists; return a NEW dict.

    Raises KeyError if the path doesn't exist (defensive — orchestration bug,
    not a runtime branch).
    """
    new = copy.deepcopy(d)
    parts = dotted.split(".")
    cursor = new
    for part in parts[:-1]:
        if isinstance(cursor, list):
            cursor = cursor[int(part)]
        else:
            cursor = cursor[part]
    last = parts[-1]
    if isinstance(cursor, list):
        cursor[int(last)] = value
    else:
        cursor[last] = value
    return new

        parquet_root
        / f"run_id={run_id}"
        / "sweep_axis=contour_2d"
        / safe_pair
    ).resolve()
    if not out_dir.is_relative_to(parquet_root_resolved):
        raise ValueError(
            f"contour_2d path traversal blocked: {out_dir} "
            f"is outside {parquet_root_resolved}"

        for row in grid for c in row
    ]
    out_path = out_dir / "results.parquet"
    pl.DataFrame(rows).write_parquet(out_path)
    return out_path

## Why reflection-driven

The BACK-08 perturbation grid enumerates `±20% individual knob perturbations`
(BACK-08 ROADMAP scope verbatim). The KEY DESIGN PROPERTY is that adding a

reflection on every call.

## Implementation idiom

`enumerate_perturbable_leaves(config)` walks `config.model_dump(mode='json')`
recursively and emits a `KnobLeaf(path, value)` for every numeric leaf:

- numeric = `int | float`, NOT `bool` (booleans handled by BACK-06
  forfeit-branch sweep, see [[Voluntary cancel reward behavior]])
- excludes `BotConfig.backtest.*` per RESEARCH §Open Question #3 — strategy
  fragility is the question; backtest-method fragility (e.g.
  `bootstrap_n_resamples`) is a different and less critical question

The reflection idiom mirrors `tests/test_manifest_contains_all_knobs.py::_collect_leaf_paths`
(lines 53-76) verbatim — `_walk(d, path)` recursion, dotted-path
concatenation, type-narrowed leaf collection. Same shape used by the manifest
writer (see [[run-manifest-writer]]); the two reflections are linked — when
adding a knob, both surface the new leaf in lockstep, and
`test_phase4_manifest_leaves.py` + `test_backtest_perturbation_grid_pydantic.py`
guard the discipline.

## Why ±20%

The 20% default (`BotConfig.backtest.perturbation_pct = 0.20`) is the
strategy-engineering convention from BACK-08 — large enough to expose
fragility, small enough that perturbed configs remain semantically valid.
Operator-overrideable.

## What "fragile" means

A knob is **fragile** if perturbing it by ±`perturbation_pct` (default 20%)
flips the sign of the bootstrapped 5th-percentile annualized P&L. Per BACK-08
ROADMAP scope verbatim: `sign-flipping on any knob is a fragility flag in
the ship-gate report.`

Sign-flip is computed via tri-valued sign extraction:

```python
sign_unp  = (unperturbed_pnl > 0) - (unperturbed_pnl < 0)
sign_low  = (low_pnl > 0)         - (low_pnl < 0)
sign_high = (high_pnl > 0)        - (high_pnl < 0)
fragile = sign_unp != sign_low or sign_unp != sign_high
```

(0.0 maps to sign 0, never spuriously flips.)

## Operator-decides discipline

**The fragility flag is RECORDED, not an automatic ship-gate fail** per
RESEARCH.md L1045. The default ship-gate decision is
`passed = (5th_percentile > threshold)` — independent of fragility flags.
Adding `AND (n_fragile_knobs == 0)` would be too strict and would miss
strategies whose marginal-but-shippable economics are flagged for legitimate
reasons (e.g. `screener.alt_yield` ±20% naturally flips a thin-margin run).

The operator inspects the `## Perturbation Grid` section of
`var/backtest/<run_id>/ship-gate-report.md` and decides whether to:

1. ship anyway (acceptable thin-margin), or
2. tighten the knob's empirical anchor (e.g. recalibrate `decision.p_purge_default`
   against contract-trace evidence), or
3. defer the strategy to v1.0 if the flag is on a known-load-bearing knob.

This mirrors the [[Voluntary cancel reward behavior]] discipline — official-team
evidence vs contract-trace evidence — same operator-judgment-not-auto-fail rule.
