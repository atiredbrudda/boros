"""BACK-02: two-axis cancel-latency simulator.

Latency has two compounding axes:
  Axis 1 (order-of-event): the bot's decision lands at the next 1 Hz tick AFTER a fill.
  Axis 2 (time-since-event): close-at-market is scheduled L seconds after fill detection.

Both apply to the same fill — they compound, they do not multiplex. PITFALLS §C2.

Realized close P&L uses underlying APR at `close_at_ts` — NOT at `fill_ts`.
The L-second window IS the inventory-at-risk period the latency simulation is meant
to capture; using fill_ts basis erases axis-2 of the latency model.

NEVER calls `time.time()` — replay engine passes blockTimestamp (chain time).
scripts/no_time_time_lint.py enforces this.

[[backtest-cancel-latency-two-axis]]
"""

from __future__ import annotations

import math
from dataclasses import dataclass


@dataclass(frozen=True)
class ScheduledClose:
    """A scheduled close-at-market event awaiting its `close_at_ts` to fire.

    Created by LatencyScheduler.schedule(...) when a synthetic order is hit by a trade.
    Consumed by ReplayEngine._fire_scheduled_closes(current_ts) when current_ts >= close_at_ts.
    """

    order_id: str
    fill_ts: float
    fill_event_index: int
    decision_ts: float
    close_at_ts: float
    fill_rate: float


class LatencyScheduler:
    """Two-axis latency. NEVER calls time.time() — replay engine passes blockTimestamp."""

    def __init__(self, L_seconds: float, decision_period_s: float = 1.0) -> None:
        if L_seconds < 0:
            raise ValueError(f"L_seconds must be >= 0; got {L_seconds}")
        if decision_period_s <= 0:
            raise ValueError(f"decision_period_s must be > 0; got {decision_period_s}")
        self.L = float(L_seconds)
        self.dt = float(decision_period_s)

    def schedule(
        self,
        fill_ts: float,
        fill_rate: float,
        order_id: str,
        *,
        fill_event_index: int = 0,
    ) -> ScheduledClose:
        """Return a ScheduledClose with two-axis latency applied.

        decision_ts = next 1 Hz tick AFTER fill_ts (axis 1).
        close_at_ts = decision_ts + L_seconds (axis 2).
        """
        # Axis 1: order-of-event. ceil to the next dt boundary.
        decision_ts = math.ceil(fill_ts / self.dt) * self.dt
        # If fill_ts is exactly on a tick (e.g., 12.0), decision_ts must be the NEXT tick (13.0)
        # because the bot processes the tick at fill_ts, then the fill, then reacts at the next tick.
        if decision_ts == fill_ts:
            decision_ts = fill_ts + self.dt
        # Axis 2: time-since-event. L seconds further.
        close_at_ts = decision_ts + self.L
        return ScheduledClose(
            order_id=order_id,
            fill_ts=fill_ts,
            fill_event_index=fill_event_index,
            decision_ts=decision_ts,
            close_at_ts=close_at_ts,
            fill_rate=fill_rate,
        )


__all__ = ["ScheduledClose", "LatencyScheduler"]
```

NOTE on the `if decision_ts == fill_ts: decision_ts = fill_ts + self.dt` clause: when `fill_ts` happens to be exactly on a tick (e.g., `fill_ts=12.0`, `dt=1.0` → `math.ceil(12.0/1.0)*1.0 = 12.0`), the bot already processed tick 12.0 BEFORE the fill landed in this block (block-ordering semantics: the decision precedes any post-tick events). So the next available decision is at the next tick. The test `test_decision_ts_is_next_one_hz_tick_after_fill` exercises both `fill_ts=11.3` (→12.0) AND `fill_ts=12.0` (→13.0) cases.

"""BACK-02 — Cancel latency two-axis simulation.

[[backtest-cancel-latency-two-axis]]
"""
from __future__ import annotations

import pytest

from boros_brain.backtest.cancel_latency import LatencyScheduler, ScheduledClose


@pytest.mark.parametrize("fill_ts,expected_decision_ts", [
    (10.5, 11.0),
    (11.3, 12.0),
    (11.9, 12.0),
    (12.0, 13.0),    # On-tick: next available decision is the NEXT tick
    (12.0001, 13.0),
])
def test_decision_ts_is_next_one_hz_tick_after_fill(fill_ts, expected_decision_ts):
    sc = LatencyScheduler(L_seconds=0.0, decision_period_s=1.0).schedule(
        fill_ts=fill_ts, fill_rate=0.05, order_id="o1",
    )
    assert sc.decision_ts == expected_decision_ts


@pytest.mark.parametrize("L,expected_close_at_ts", [
    (5.0, 17.0),
    (2.0, 14.0),
    (0.5, 12.5),
    (10.0, 22.0),
])
def test_close_at_ts_equals_decision_ts_plus_L(L, expected_close_at_ts):
    sc = LatencyScheduler(L_seconds=L, decision_period_s=1.0).schedule(
        fill_ts=11.3, fill_rate=0.05, order_id="o1",
    )
    assert sc.close_at_ts == expected_close_at_ts


@pytest.mark.parametrize("fill_ts,L,expected_inventory_at_risk", [
    (11.3, 5.0, 5.7),    # axis1 = 0.7s, axis2 = 5.0s, total 5.7s (compounded)
    (11.0, 2.0, 3.0),    # 11.0 → 12.0 (1.0s) + 2.0s = 3.0s
    (12.0, 5.0, 6.0),    # on-tick: 12.0 → 13.0 (1.0s) + 5.0s = 6.0s
])
def test_two_axes_compound_not_multiplex(fill_ts, L, expected_inventory_at_risk):
    sc = LatencyScheduler(L_seconds=L, decision_period_s=1.0).schedule(
        fill_ts=fill_ts, fill_rate=0.05, order_id="o1",
    )
    total = (sc.decision_ts - sc.fill_ts) + L  # SUM, not MAX
    assert abs(total - expected_inventory_at_risk) < 1e-9


def test_validation_rejects_negative_L():
    with pytest.raises(ValueError, match="L_seconds must be >= 0"):
        LatencyScheduler(L_seconds=-0.1, decision_period_s=1.0)


def test_validation_rejects_zero_decision_period():
    with pytest.raises(ValueError, match="decision_period_s must be > 0"):

               fired.append(sc)
           else:
               remaining.append(sc)
       self._scheduled_closes = remaining

   def _underlying_at(self, ts: float) -> float:
       """Read latest underlying_apr row with snapshot_ts <= ts."""
       try:
           df = (
               pl.scan_parquet(str(self.parquet_root / "underlying_apr" / "**" / "*.parquet"))
               .filter(pl.col("snapshot_ts") <= int(ts))
               .sort("snapshot_ts")
               .tail(1)
               .collect()
           )
           if df.height == 0:
               return 0.05  # default
           return float(df.select("rate").item())
       except Exception:
           return 0.05
   ```
