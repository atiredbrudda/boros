```python
"""Pydantic + dataclass shapes for the backtest engine (BACK-01..08).

Single home for all Phase-5 shape declarations so architecture.yml has a single
brain.backtest.schemas node.

[[backtest-output-schema-and-parquet-partitioning]]
"""

from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Literal

import numpy as np
from pydantic import BaseModel, ConfigDict


class StrictBase(BaseModel):
    """Local strict base — extra='forbid', frozen=True."""

    model_config = ConfigDict(extra="forbid", frozen=True)


class BacktestRunOutput(StrictBase):
    """One row per (run_id, sweep_axis, sweep_value, bootstrap_idx)."""

    # Provenance
    run_id: str
    parent_run_id: str | None = None
    git_sha: str
    strategy_version: str
    market_id: str
    side: Literal["long", "short"]

    # Sweep coordinates
    sweep_axis: Literal["base", "cancel_latency", "competition", "forfeit_branch", "perturbation"]
    sweep_value: float | str
    bootstrap_idx: int | None = None

    # Core P&L
    raw_pnl_pendle: float
    raw_pnl_usd: float
    annualized_pnl_pct: float
    sharpe_equivalent: float
    max_drawdown_pct: float

    # Activity
    n_decisions: int
    n_fills: int
    n_cancels: int
    n_purges: int
    cumulative_fees_pendle: float
    cumulative_gas_pendle: float

    # Calibration
    predicted_p_fill_mean: float
    realized_fill_rate: float
    calibration_residual: float

    # Lineage flags
    e_liq_data_source: Literal["operator"] = "operator"
    e_liq_calibrated_at: str
    event_index_coverage: Literal["full", "degraded"]
    voluntary_cancel_branch: Literal["benign", "punitive"]

    # Output paths
    parquet_partition: str
    plots_dir: str | None = None


@dataclass(frozen=True)
class ReplayConfig:
    """In-memory configuration for a single ReplayEngine.run() invocation."""

    config: Any  # BotConfig — use Any to avoid circular import
    market_id: str
    side: Literal["long", "short"]
    parquet_root: Path
    sqlite_path: Path
    cancel_latency_s: float
    voluntary_cancel_branch: Literal["benign", "punitive"] = "benign"
    sigma_others_multiplier: float = 1.0


@dataclass
class ReplayOutput:
    """Output of a single ReplayEngine.run() invocation. Mutable: built up incrementally."""

    rows: list[BacktestRunOutput] = field(default_factory=list)
    decision_log: list[dict[str, Any]] = field(default_factory=list)
    daily_pnl: np.ndarray = field(default_factory=lambda: np.empty(0))
    event_index_coverage: Literal["full", "degraded"] = "full"
    n_amm_violations: int = 0  # MUST stay 0; violations raise AMMConsumptionIdentityError


__all__ = ["StrictBase", "BacktestRunOutput", "ReplayConfig", "ReplayOutput"]
```

```python
"""BACK-01: in-memory synthetic-order ledger.

Mirrors state.open_orders shape but lives in-memory only — backtest is hermetic
and never writes to SQLite (single-writer-per-table discipline; backtest is a
read-only-on-state phase per RESEARCH.md L514-524).

Defensive: cancel/fill/purge raise KeyError on unknown order_id (orchestration
bugs surface loudly, not silently).
"""

from __future__ import annotations

from dataclasses import dataclass, replace
from typing import Iterator, Literal


@dataclass(frozen=True)
class SyntheticOrder:
    order_id: str
    market_id: str
    side: Literal["long", "short"]
    tick: int
    q_me_pendle: float
    placed_at_ts: int
    depth_from_mark_ticks: int
    queue_ahead_wei: int = 0
    r_accum: float = 0.0
    status: Literal["active", "filled", "cancelled", "purged"] = "active"
    t_frac: float = 1.0


class SyntheticOrderLedger:
    """In-memory order book mirror. NEVER persisted; backtest is hermetic."""

    def __init__(self) -> None:
        self._orders: dict[str, SyntheticOrder] = {}

    def place(self, order: SyntheticOrder) -> None:
        self._orders[order.order_id] = order

    def cancel(self, order_id: str) -> None:
        self._orders[order_id] = replace(self._orders[order_id], status="cancelled")

    def mark_filled(self, order_id: str) -> None:
        self._orders[order_id] = replace(self._orders[order_id], status="filled")

    def mark_purged(self, order_id: str) -> None:
        self._orders[order_id] = replace(self._orders[order_id], status="purged")

    def update_queue_ahead(self, order_id: str, queue_ahead_wei: int) -> None:
        self._orders[order_id] = replace(
            self._orders[order_id], queue_ahead_wei=queue_ahead_wei
        )

    def in_flight(self) -> Iterator[SyntheticOrder]:
        return (o for o in self._orders.values() if o.status == "active")

    def get(self, order_id: str) -> SyntheticOrder | None:
        return self._orders.get(order_id)


__all__ = ["SyntheticOrder", "SyntheticOrderLedger"]
```
